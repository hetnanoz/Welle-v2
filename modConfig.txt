Option Explicit

Private Const CLASS_NAME As String = "modConfig"

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-27
' Parameters:    appConfig As TAppConfig
' Returns:       ---
' Description:   Reads and validates all named configuration cells.
'-------------------------------------------------------------------------------
Public Sub LoadConfiguration(ByRef appConfig As TAppConfig)
    Const METHOD_NAME As String = "LoadConfiguration"
    Dim errDescription As String
    Dim errNumber As Long
    Dim strConfigName As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    strConfigName = NAME_FLAG_SAVE_LOCAL
    appConfig.SaveLocal = ReadBooleanFlag(strConfigName)
    strConfigName = NAME_FLAG_SAVE_SHAREPOINT
    appConfig.SaveSharePoint = ReadBooleanFlag(strConfigName)
    strConfigName = NAME_FLAG_UPDATE_HISTORICAL
    appConfig.UpdateHistorical = ReadBooleanFlag(strConfigName)
    strConfigName = NAME_FLAG_CREATE_TEAM_FILES
    appConfig.CreateTeamFiles = ReadBooleanFlag(strConfigName)
    strConfigName = NAME_FLAG_ARCHIVE_INPUT_FILES
    appConfig.ArchiveInputFiles = ReadBooleanFlag(strConfigName)
    strConfigName = NAME_FLAG_ARCHIVE_OUTLOOK_MESSAGES
    appConfig.ArchiveOutlookMessages = ReadBooleanFlag(strConfigName)

    strConfigName = NAME_OUTLOOK_MAILBOX
    appConfig.OutlookMailbox = ReadRequiredText(strConfigName)
    strConfigName = NAME_OUTLOOK_SOURCE_FOLDER
    appConfig.OutlookSourceFolder = ReadRequiredText(strConfigName)
    strConfigName = NAME_OUTLOOK_ARCHIVE_FOLDER
    If appConfig.ArchiveOutlookMessages Then
        appConfig.OutlookArchiveFolder = ReadRequiredText(strConfigName)
    Else
        appConfig.OutlookArchiveFolder = ReadOptionalText(strConfigName)
    End If

    ' Optional ID pairs skip expensive mailbox/folder traversal when valid.
    strConfigName = NAME_OUTLOOK_SOURCE_ENTRY_ID
    appConfig.OutlookSourceEntryId = ReadOptionalTextIfDefined(strConfigName)
    strConfigName = NAME_OUTLOOK_SOURCE_STORE_ID
    appConfig.OutlookSourceStoreId = ReadOptionalTextIfDefined(strConfigName)
    strConfigName = NAME_OUTLOOK_ARCHIVE_ENTRY_ID
    appConfig.OutlookArchiveEntryId = ReadOptionalTextIfDefined(strConfigName)
    strConfigName = NAME_OUTLOOK_ARCHIVE_STORE_ID
    appConfig.OutlookArchiveStoreId = ReadOptionalTextIfDefined(strConfigName)

    strConfigName = NAME_OUTLOOK_SUBJECT_PREFIX
    appConfig.OutlookSubjectPrefix = ReadRequiredText(strConfigName)
    strConfigName = NAME_OUTLOOK_SENDER_EMAIL
    appConfig.OutlookSenderEmail = ReadOptionalText(strConfigName)
    strConfigName = NAME_OUTLOOK_ATTACHMENT_PREFIX
    appConfig.OutlookAttachmentPrefix = ReadRequiredText(strConfigName)
    strConfigName = NAME_INPUT_WORKSHEET_NAME
    appConfig.InputWorksheetName = ReadRequiredText(strConfigName)
    strConfigName = NAME_FONDSLISTE_FOLDER
    appConfig.FondslisteFolder = ReadRequiredText(strConfigName)
    ' New typo-compatible Mapping name takes priority; legacy name still works.
    strConfigName = NAME_FONDSLISTE_PARREN
    If Not IsMappingNameDefined(strConfigName) Then
        strConfigName = NAME_FONDSLISTE_PATTERN
    End If
    ' In FIXED mode this value is an exact base name, never a wildcard pattern.
    appConfig.FondslistePattern = ReadRequiredText(strConfigName)
    strConfigName = NAME_FONDSLISTE_EXTENSION
    appConfig.FondslisteExtension = ReadRequiredText(strConfigName)
    strConfigName = NAME_FLAG_FONDSLISTE_LOCAL_COPY
    appConfig.FondslisteLocalCopy = ReadOptionalBooleanFlagIfDefined(strConfigName)
    strConfigName = NAME_SUPPORTED_TEAMS
    appConfig.SupportedTeams = ReadRequiredText(strConfigName)

    strConfigName = NAME_CREATE_EMAIL_DRAFT
    appConfig.CreateEmailDraft = ReadOptionalBooleanFlag(strConfigName)

    If appConfig.CreateEmailDraft Then
        strConfigName = NAME_MAIL_SUBJECT
        appConfig.MailSubject = ReadRequiredText(strConfigName)
        strConfigName = NAME_MAIL_BODY
        appConfig.MailBody = ReadOptionalText(strConfigName)
        strConfigName = NAME_MAIL_BODY_NO_TEAMS
        appConfig.MailBodyNoTeams = ReadOptionalText(strConfigName)
        strConfigName = NAME_MAIL_BODY_COMBINED_ONLY
        appConfig.MailBodyCombinedOnly = ReadOptionalTextIfDefined(strConfigName)
        strConfigName = NAME_MAIL_TO
        appConfig.MailTo = ReadRequiredText(strConfigName)
        strConfigName = NAME_MAIL_CC
        appConfig.MailCc = ReadOptionalText(strConfigName)
        strConfigName = NAME_MAIL_FROM
        appConfig.MailFrom = ReadOptionalText(strConfigName)
        strConfigName = NAME_MAIL_SHAREPOINT_URL
        appConfig.MailSharePointUrl = ReadOptionalText(strConfigName)
    End If

    If appConfig.SaveLocal Then
        strConfigName = NAME_OUTPUT_LOCAL_BASE
        appConfig.OutputLocalBase = ReadRequiredText(strConfigName)
    End If

    If appConfig.SaveSharePoint Then
        strConfigName = NAME_OUTPUT_SHAREPOINT_BASE
        appConfig.OutputSharePointBase = ReadRequiredText(strConfigName)
    End If

    strConfigName = "cross-field validation"
    Call ValidateConfiguration(appConfig)

ExitPoint:
    If errNumber <> 0 Then
        Call VBA.Err.Raise( _
            errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    If Len(strConfigName) > 0 Then
        errDescription = _
            "Configuration item: " & strConfigName & ". " & errDescription
    End If
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "configName", strConfigName)
    GoTo ExitPoint
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-29
' Parameters:    strName - configured Mapping name
' Returns:       Boolean
' Description:   Checks whether a named setting exists. If it exists but does
'                not refer to a Mapping cell, ReadRequiredText raises an error.
'-------------------------------------------------------------------------------
Private Function IsMappingNameDefined(ByVal strName As String) As Boolean
    Const METHOD_NAME As String = "IsMappingNameDefined"
    Dim blnFound As Boolean
    Dim errDescription As String
    Dim errNumber As Long
    Dim objName As Excel.Name

    If Not DEV_MODE Then On Error GoTo ErrHandler

    For Each objName In ThisWorkbook.Names
        If StrComp(GetUnqualifiedName(objName.Name), strName, vbTextCompare) = 0 Then
            blnFound = True
            Exit For
        End If
    Next objName

ExitPoint:
    Set objName = Nothing
    If errNumber = 0 Then IsMappingNameDefined = blnFound
    If errNumber <> 0 Then
        Call VBA.Err.Raise( _
            errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "name", strName)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-26
' Parameters:    strName As String
' Returns:       Excel.Range
' Description:   Returns a one-cell named range located on Mapping.
'-------------------------------------------------------------------------------
Private Function GetNamedCellRange(ByVal strName As String) As Excel.Range
    Const METHOD_NAME As String = "GetNamedCellRange"
    Dim errDescription As String
    Dim errNumber As Long
    Dim objName As Excel.Name
    Dim rngCandidate As Excel.Range
    Dim rngResult As Excel.Range
    Dim strRefersTo As String
    Dim strSimpleName As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    For Each objName In ThisWorkbook.Names
        strSimpleName = GetUnqualifiedName(objName.Name)

        If StrComp(strSimpleName, strName, vbTextCompare) = 0 Then
            strRefersTo = CStr(objName.RefersTo)

            If InStr(1, strRefersTo, "#REF!", vbTextCompare) > 0 Then
                Call VBA.Err.Raise( _
                    ERROR_CONFIGURATION, METHOD_NAME, _
                    "Named configuration cell is broken (#REF!): " & strName & _
                    ". Reassign it in Formulas > Name Manager.")
            End If

            If InStr(1, strRefersTo, "!", vbBinaryCompare) = 0 Then
                Call VBA.Err.Raise( _
                    ERROR_CONFIGURATION, METHOD_NAME, _
                    "Named configuration item does not refer to a worksheet cell: " & _
                    strName)
            End If

            Set rngCandidate = objName.RefersToRange

            If StrComp( _
                CStr(rngCandidate.Parent.Name), SHEET_MAPPING, vbTextCompare) = 0 Then

                Set rngResult = rngCandidate
                Exit For
            End If
        End If
    Next objName

    If rngResult Is Nothing Then
        Call VBA.Err.Raise( _
            ERROR_CONFIGURATION, METHOD_NAME, _
            "Named cell was not found on Mapping: " & strName)
    End If

    If rngResult.Cells.CountLarge <> 1 Then
        Call VBA.Err.Raise( _
            ERROR_CONFIGURATION, METHOD_NAME, _
            "Named range must refer to exactly one cell: " & strName)
    End If

ExitPoint:
    Set rngCandidate = Nothing
    Set objName = Nothing
    If errNumber = 0 Then Set GetNamedCellRange = rngResult
    Set rngResult = Nothing

    If errNumber <> 0 Then
        Call VBA.Err.Raise( _
            errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description

    If errNumber = 1004 Then
        errDescription = _
            "Named configuration item exists but does not resolve to a valid " & _
            "cell on Mapping: " & strName & ". Check Formulas > Name Manager " & _
            "for #REF!, wrong scope or a non-cell reference. Original error: " & _
            errDescription
    End If

    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "name", strName)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-26
' Parameters:    strQualifiedName As String
' Returns:       String
' Description:   Removes workbook or worksheet qualification from a name.
'-------------------------------------------------------------------------------
Private Function GetUnqualifiedName(ByVal strQualifiedName As String) As String
    Const METHOD_NAME As String = "GetUnqualifiedName"
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngPosition As Long
    Dim strResult As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    lngPosition = InStrRev(strQualifiedName, "!")

    If lngPosition > 0 Then
        strResult = Mid$(strQualifiedName, lngPosition + 1)
    Else
        strResult = strQualifiedName
    End If

    strResult = Replace$(strResult, "'", vbNullString)

ExitPoint:
    If errNumber = 0 Then GetUnqualifiedName = strResult
    If errNumber <> 0 Then Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "qualifiedName", strQualifiedName)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-26
' Parameters:    strName As String
' Returns:       String
' Description:   Reads a required text value from a named Mapping cell.
'-------------------------------------------------------------------------------
Private Function ReadRequiredText(ByVal strName As String) As String
    Const METHOD_NAME As String = "ReadRequiredText"
    Dim errDescription As String
    Dim errNumber As Long
    Dim rngValue As Excel.Range
    Dim strResult As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set rngValue = GetNamedCellRange(strName)
    strResult = Trim$(CStr(rngValue.Value2))

    If Len(strResult) = 0 Then
        Call VBA.Err.Raise( _
            ERROR_CONFIGURATION, METHOD_NAME, _
            "Required configuration value is blank: " & strName)
    End If

ExitPoint:
    Set rngValue = Nothing
    If errNumber = 0 Then ReadRequiredText = strResult
    If errNumber <> 0 Then Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError(CLASS_NAME, METHOD_NAME, errNumber, errDescription, "name", strName)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-26
' Parameters:    strName As String
' Returns:       Boolean
' Description:   Reads a Boolean flag from a named Mapping cell.
'-------------------------------------------------------------------------------
Private Function ReadBooleanFlag(ByVal strName As String) As Boolean
    Const METHOD_NAME As String = "ReadBooleanFlag"
    Dim blnResult As Boolean
    Dim errDescription As String
    Dim errNumber As Long
    Dim rngValue As Excel.Range
    Dim strValue As String
    Dim varValue As Variant

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set rngValue = GetNamedCellRange(strName)
    varValue = rngValue.Value2

    If VarType(varValue) = vbBoolean Then
        blnResult = CBool(varValue)
    ElseIf IsNumeric(varValue) Then
        blnResult = CDbl(varValue) <> 0
    Else
        strValue = UCase$(Trim$(CStr(varValue)))
        strValue = Replace$(strValue, ChrW(321), "L")
        strValue = Replace$(strValue, ChrW(322), "L")

        Select Case strValue
            Case "TRUE", "PRAWDA", "YES", "Y", "1"
                blnResult = True
            Case "FALSE", "FALSZ", "NO", "N", "0"
                blnResult = False
            Case Else
                Call VBA.Err.Raise( _
                    ERROR_CONFIGURATION, METHOD_NAME, _
                    "Invalid Boolean value in named cell: " & strName)
        End Select
    End If

ExitPoint:
    Set rngValue = Nothing
    If errNumber = 0 Then ReadBooleanFlag = blnResult
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "name", strName)
    GoTo ExitPoint
End Function
 '-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-27
' Parameters:    strName As String
' Returns:       String
' Description:   Reads an optional text value from a named Mapping cell.
'-------------------------------------------------------------------------------
Private Function ReadOptionalText(ByVal strName As String) As String

    Dim errNumber As Long
    Dim rngValue As Excel.Range

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set rngValue = GetNamedCellRange(strName)

    If Not IsError(rngValue.Value2) Then
        If Not IsNull(rngValue.Value2) And Not IsEmpty(rngValue.Value2) Then
            strResult = Trim$(CStr(rngValue.Value2))
        End If
    End If

ExitPoint:
    Set rngValue = Nothing
    If errNumber = 0 Then ReadOptionalText = strResult
    If errNumber <> 0 Then Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError(CLASS_NAME, METHOD_NAME, errNumber, errDescription, "name", strName)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-26
' Parameters:    appConfig As TAppConfig
' Returns:       ---
' Description:   Validates cross-field configuration requirements.
'-------------------------------------------------------------------------------
Private Sub ValidateConfiguration(ByRef appConfig As TAppConfig)
    Const METHOD_NAME As String = "ValidateConfiguration"
    Dim errDescription As String
    Dim errNumber As Long

    If Not DEV_MODE Then On Error GoTo ErrHandler

    If Not appConfig.SaveLocal And Not appConfig.SaveSharePoint Then
        Call VBA.Err.Raise( _
            ERROR_CONFIGURATION, METHOD_NAME, _
            "At least one output flag must be TRUE.")
    End If

    If appConfig.SaveLocal And Len(Trim$(appConfig.OutputLocalBase)) = 0 Then
        Call VBA.Err.Raise( _
            ERROR_CONFIGURATION, METHOD_NAME, _
            "Local output path is required when FLAG_SAVE_LOCAL is TRUE.")
    End If

    If appConfig.SaveSharePoint And Len(Trim$(appConfig.OutputSharePointBase)) = 0 Then
        Call VBA.Err.Raise( _
            ERROR_CONFIGURATION, METHOD_NAME, _
            "SharePoint output path is required when FLAG_SAVE_SHAREPOINT is TRUE.")
    End If

    If Not HasNonBlankSemicolonOption(appConfig.OutlookSubjectPrefix) Then
        Call VBA.Err.Raise( _
            ERROR_CONFIGURATION, METHOD_NAME, _
            "OUTLOOK_SUBJECT_PREFIX must contain at least one value. " & _
            "Multiple alternatives may be separated with ';'.")
    End If

    If InStr(1, appConfig.OutlookSubjectPrefix, "*", vbBinaryCompare) > 0 Then
        Call VBA.Err.Raise( _
            ERROR_CONFIGURATION, METHOD_NAME, _
            "OUTLOOK_SUBJECT_PREFIX must not contain '*'. " & _
            "Enter one or more subject phrases separated with ';'.")
    End If

    If Len(Trim$(appConfig.OutlookSenderEmail)) > 0 Then
        If Not HasNonBlankSemicolonOption(appConfig.OutlookSenderEmail) Then
            Call VBA.Err.Raise( _
                ERROR_CONFIGURATION, METHOD_NAME, _
                "OUTLOOK_SENDER_EMAIL must contain at least one email address " & _
                "or be blank. Multiple alternatives may be separated with ';'.")
        End If
    End If

    If Not HasNonBlankSemicolonOption(appConfig.OutlookAttachmentPrefix) Then
        Call VBA.Err.Raise( _
            ERROR_CONFIGURATION, METHOD_NAME, _
            "OUTLOOK_ATTACHMENT_PREFIX must contain at least one value. " & _
            "Multiple alternatives may be separated with ';'.")
    End If

    If InStr(1, appConfig.OutlookAttachmentPrefix, "*", vbBinaryCompare) > 0 Then
        Call VBA.Err.Raise( _
            ERROR_CONFIGURATION, METHOD_NAME, _
            "OUTLOOK_ATTACHMENT_PREFIX must not contain '*'. Enter one or more " & _
            "attachment filename prefixes separated with ';'.")
    End If

    If Not HasNonBlankSemicolonOption(appConfig.InputWorksheetName) Then
        Call VBA.Err.Raise( _
            ERROR_CONFIGURATION, METHOD_NAME, _
            "INPUT_WORKSHEET_NAME must contain at least one worksheet name. " & _
            "Multiple alternatives may be separated with ';'.")
    End If

    If appConfig.ArchiveOutlookMessages Then
        If Len(Trim$(appConfig.OutlookArchiveFolder)) = 0 Then
            Call VBA.Err.Raise( _
                ERROR_CONFIGURATION, METHOD_NAME, _
                "OUTLOOK_ARCHIVE_FOLDER is required when " & _
                "FLAG_ARCHIVE_OUTLOOK_MESSAGES is TRUE.")
        End If

        If StrComp( _
            appConfig.OutlookSourceFolder, appConfig.OutlookArchiveFolder, _
            vbTextCompare) = 0 Then

            Call VBA.Err.Raise( _
                ERROR_CONFIGURATION, METHOD_NAME, _
                "Outlook source and archive folders must be different.")
        End If
    End If

    If appConfig.CreateEmailDraft Then
        If Len(Trim$(appConfig.MailSubject)) = 0 Then
            Call VBA.Err.Raise( _
                ERROR_CONFIGURATION, METHOD_NAME, _
                "MAIL_SUBJECT is required when CREATE_EMAIL_DRAFT is TRUE.")
        End If

        If Len(Trim$(appConfig.MailTo)) = 0 Then
            Call VBA.Err.Raise( _
                ERROR_CONFIGURATION, METHOD_NAME, _
                "MAIL_TO is required when CREATE_EMAIL_DRAFT is TRUE.")
        End If
    End If

ExitPoint:
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-20
' Parameters:    strOptions As String
' Returns:       Boolean
' Description:   Confirms that a semicolon-separated Mapping value contains at least one non-blank option.
'-------------------------------------------------------------------------------
Private Function HasNonBlankSemicolonOption(ByVal strOptions As String) As Boolean
    Const METHOD_NAME As String = "HasNonBlankSemicolonOption"
    Dim arrOptions As Variant
    Dim blnResult As Boolean
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngOption As Long

    If Not DEV_MODE Then On Error GoTo ErrHandler

    arrOptions = Split(strOptions, ";")

    For lngOption = LBound(arrOptions) To UBound(arrOptions)
        If Len(Trim$(CStr(arrOptions(lngOption)))) > 0 Then
            blnResult = True
            Exit For
        End If
    Next lngOption

ExitPoint:
    If errNumber = 0 Then HasNonBlankSemicolonOption = blnResult
    If errNumber <> 0 Then Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Function
