'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-27
' Parameters:    Configuration, combined data, Teams, run timestamp, label result.
' Returns:       ---
' Description:   Creates and saves an Outlook draft for the completed daily process.
'-------------------------------------------------------------------------------
Public Sub CreateCashTransactionsEmailDraft( _
    ByRef appConfig As TAppConfig, _
    ByRef arrCombined As Variant, _
    ByVal colTeams As Collection, _
    ByVal dtRunTimestamp As Date, _
    ByRef blnLabelApplied As Boolean)
    Const METHOD_NAME As String = "CreateCashTransactionsEmailDraft"
    Dim errDescription As String
    Dim errNumber As Long
    Dim objOutlook As Object
    Dim objMail As Object
    Dim strBodyHtml As String
    Dim strLocalFolder As String
    Dim strSharePointFolder As String
    Dim strSharePointUrl As String
    Dim strSignatureHtml As String
    Dim strSubject As String
    Dim strTeamList As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    strTeamList = BuildEmailTeamList(arrCombined, colTeams)
    strSubject = BuildEmailSubject( _
        appConfig.MailSubject, dtRunTimestamp, strTeamList, _
        appConfig.CreateTeamFiles)
    strLocalFolder = GetEmailLocalFolder(appConfig, dtRunTimestamp)
    strSharePointUrl = GetEmailSharePointUrl(appConfig)

    If appConfig.SaveSharePoint And Len(Trim$(strSharePointUrl)) = 0 Then
        strSharePointFolder = GetEmailSharePointFolder(appConfig, dtRunTimestamp)
    End If

    strBodyHtml = BuildEmailBodyHtml( _
        appConfig, arrCombined, dtRunTimestamp, strTeamList, _
        strLocalFolder, strSharePointUrl, strSharePointFolder)

    Set objOutlook = CreateObject("Outlook.Application")
    Set objMail = objOutlook.CreateItem(0)

    If Len(Trim$(appConfig.MailFrom)) > 0 Then objMail.SentOnBehalfOfName = appConfig.MailFrom
    objMail.To = appConfig.MailTo
    objMail.Cc = appConfig.MailCc
    objMail.Subject = strSubject
    objMail.BodyFormat = 2  ' olFormatHTML; keep late binding.

    objMail.Display
    strSignatureHtml = CStr(objMail.HTMLBody)
    objMail.HTMLBody = _
        "<div style='font-family:Arial;font-size:11pt;'>" & _
        strBodyHtml & "</div><br>" & strSignatureHtml

    blnLabelApplied = TryApplyInternalLabel(objMail)
    objMail.Save

ExitPoint:
    Set objMail = Nothing
    Set objOutlook = Nothing

    If errNumber <> 0 Then Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-27
' Parameters:    arrCombined As Variant; colTeams As Collection
' Returns:       String
' Description:   Returns configured Teams that have at least one transaction in the current run.
'-------------------------------------------------------------------------------
Private Function BuildEmailTeamList( _
    ByRef arrCombined As Variant, _
    ByVal colTeams As Collection) As String
    Const METHOD_NAME As String = "BuildEmailTeamList"
    Dim dictTeamsWithTrades As Object
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngRow As Long
    Dim lngTeam As Long
    Dim strCurrentTeam As String
    Dim strResult As String
    Dim strTeam As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set dictTeamsWithTrades = CreateObject("Scripting.Dictionary")
    dictTeamsWithTrades.CompareMode = vbTextCompare

    For lngRow = 2 To UBound(arrCombined, 1)
        strCurrentTeam = Trim$(CStr(arrCombined(lngRow, 5)))
        If Len(strCurrentTeam) > 0 Then dictTeamsWithTrades(strCurrentTeam) = True
    Next lngRow

    For lngTeam = 1 To colTeams.Count
        strTeam = Trim$(CStr(colTeams(lngTeam)))

        If Len(strTeam) > 0 Then
            If dictTeamsWithTrades.Exists(strTeam) Then
                If Len(strResult) > 0 Then strResult = strResult & ", "
                strResult = strResult & strTeam
            End If
        End If
    Next lngTeam

ExitPoint:
    Set dictTeamsWithTrades = Nothing
    If errNumber = 0 Then BuildEmailTeamList = strResult
    If errNumber <> 0 Then Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "row", lngRow)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-27
' Parameters:    Base subject, run date, Team list and Team-file mode.
' Returns:       String
' Description:   Builds a Team-aware or Combined-only Outlook subject.
'-------------------------------------------------------------------------------
Private Function BuildEmailSubject( _
    ByVal strBaseSubject As String, _
    ByVal dtRunTimestamp As Date, _
    ByVal strTeamList As String, _
    ByVal blnCreateTeamFiles As Boolean) As String

    Const METHOD_NAME As String = "BuildEmailSubject"
    Dim errDescription As String
    Dim errNumber As Long
    Dim strResult As String
    Dim strStatus As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    If blnCreateTeamFiles Then
        If Len(strTeamList) > 0 Then
            strStatus = "Team " & strTeamList
        Else
            strStatus = "no teams"
        End If
    Else
        strStatus = "Combined"
    End If

    strResult = _
        Trim$(strBaseSubject) & " " & _
        Format$(dtRunTimestamp, "dd/mm/yyyy") & " / " & strStatus

ExitPoint:
    If errNumber = 0 Then BuildEmailSubject = strResult
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "createTeamFiles;hasTeams", blnCreateTeamFiles, Len(strTeamList) > 0)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-27
' Parameters:    Configuration, combined data, run timestamp, Teams, output locations.
' Returns:       String
' Description:   Builds an HTML email body from configured text and supported dynamic tokens.
'-------------------------------------------------------------------------------
Private Function BuildEmailBodyHtml( _
    ByRef appConfig As TAppConfig, _
    ByRef arrCombined As Variant, _
    ByVal dtRunTimestamp As Date, _
    ByVal strTeamList As String, _
    ByVal strLocalFolder As String, _
    ByVal strSharePointUrl As String, _
    ByVal strSharePointFolder As String) As String
    Const METHOD_NAME As String = "BuildEmailBodyHtml"
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngTransactionCount As Long
    Dim lngUnknownCount As Long
    Dim rngTemplate As Excel.Range
    Dim strOutputLocationsHtml As String
    Dim strHistoricalFolderHtml As String
    Dim strResult As String
    Dim strSelectedName As String
    Dim strTeamsToken As String
    Dim strTemplate As String
    Dim strUnknownStatus As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    lngTransactionCount = UBound(arrCombined, 1) - 1
    lngUnknownCount = CountUnknownTransactions(arrCombined)
    strUnknownStatus = BuildUnknownStatus( _
        lngUnknownCount, appConfig.CreateTeamFiles, appConfig.UpdateHistorical)

    If Not appConfig.CreateTeamFiles Then
        strSelectedName = NAME_MAIL_BODY_COMBINED_ONLY
        strTeamsToken = "all teams"
        strTemplate = appConfig.MailBodyCombinedOnly
        If Len(strTemplate) = 0 Then strTemplate = GetCombinedOnlyMailBody()
    ElseIf Len(strTeamList) > 0 Then
        strSelectedName = NAME_MAIL_BODY
        strTeamsToken = strTeamList
        strTemplate = appConfig.MailBody
        If Len(strTemplate) = 0 Then strTemplate = GetDefaultMailBody(True)
    Else
        strSelectedName = NAME_MAIL_BODY_NO_TEAMS
        strTeamsToken = "no teams"
        strTemplate = appConfig.MailBodyNoTeams
        If Len(strTemplate) = 0 Then strTemplate = GetDefaultMailBody(False)
    End If

    ' Match the chosen Mapping cell to the text already validated by modConfig.
    ' Built-in fallback bodies remain plain when no Mapping text was provided.
    Set rngTemplate = GetMappingMailTemplateCell(strSelectedName, strTemplate)
    strOutputLocationsHtml = BuildOutputLocationsHtml( _
        appConfig, strLocalFolder, strSharePointUrl, strSharePointFolder)

    If InStr(1, strTemplate, "[MASTER_HISTORICAL_FOLDER]", vbTextCompare) > 0 Then
        strHistoricalFolderHtml = BuildMasterHistoricalFolderHtml( _
            appConfig.MasterHistoricalFolderPath)
    End If

    strResult = RenderMailTemplateHtml( _
        strTemplate, rngTemplate, Format$(dtRunTimestamp, "dd/mm/yyyy"), _
        strTeamsToken, strLocalFolder, lngTransactionCount, _
        lngUnknownCount, strUnknownStatus, strOutputLocationsHtml, _
        strHistoricalFolderHtml)

ExitPoint:
    Set rngTemplate = Nothing
    If errNumber = 0 Then BuildEmailBodyHtml = strResult
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "selectedTemplate", strSelectedName)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-29
' Parameters:    strName As String; strExpectedText As String
' Returns:       Excel.Range or Nothing
' Description:   Resolves the selected one-cell Mapping template for rich-text access.
'-------------------------------------------------------------------------------
Private Function GetMappingMailTemplateCell( _
    ByVal strName As String, _
    ByVal strExpectedText As String) As Excel.Range
    Const METHOD_NAME As String = "GetMappingMailTemplateCell"
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngSeparator As Long
    Dim objName As Excel.Name
    Dim rngCandidate As Excel.Range
    Dim rngResult As Excel.Range
    Dim strCellText As String
    Dim strSimpleName As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    For Each objName In ThisWorkbook.Names
        strSimpleName = CStr(objName.Name)
        lngSeparator = InStrRev(strSimpleName, "!")
        If lngSeparator > 0 Then
            strSimpleName = Mid$(strSimpleName, lngSeparator + 1)
        End If
        strSimpleName = Replace$(strSimpleName, "'", vbNullString)

        If StrComp(strSimpleName, strName, vbTextCompare) = 0 Then
            Set rngCandidate = objName.RefersToRange

            If rngCandidate.Cells.CountLarge = 1 Then
                If StrComp(CStr(rngCandidate.Parent.Name), SHEET_MAPPING, vbTextCompare) = 0 Then
                    If Not IsError(rngCandidate.Value2) Then
                        strCellText = Trim$(CStr(rngCandidate.Value2))
                        If StrComp(strCellText, strExpectedText, vbBinaryCompare) = 0 Then
                            Set rngResult = rngCandidate
                            Exit For
                        End If
                    End If
                End If
            End If
            Set rngCandidate = Nothing
        End If
    Next objName

ExitPoint:
    If errNumber = 0 Then Set GetMappingMailTemplateCell = rngResult
    Set rngCandidate = Nothing
    Set rngResult = Nothing
    Set objName = Nothing
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "name", strName)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-29
' Parameters:    Mapping body, source cell and dynamic replacement values.
' Returns:       String
' Description:   Preserves Excel character formatting while safely generating HTML.
'-------------------------------------------------------------------------------
Private Function RenderMailTemplateHtml( _
    ByVal strTemplate As String, _
    ByVal rngTemplate As Excel.Range, _
    ByVal strRunDate As String, _
    ByVal strTeams As String, _
    ByVal strLocalFolder As String, _
    ByVal lngTransactionCount As Long, _
    ByVal lngUnknownCount As Long, _
    ByVal strUnknownStatus As String, _
    ByVal strOutputLocationsHtml As String, _
    ByVal strHistoricalFolderHtml As String) As String
    Const METHOD_NAME As String = "RenderMailTemplateHtml"
    Const LEGACY_HEADER As String = "Files are available on SharePoint:"
    Const NEW_HEADER As String = "Files are available in the following location(s):"
    Const LEGACY_SHAREPOINT_LINE As String = "SharePoint: [SHAREPOINT_URL]"
    Dim blnHasOutputToken As Boolean
    Dim blnUseRichText As Boolean
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngCellOffset As Long
    Dim lngConsumed As Long
    Dim lngPosition As Long
    Dim strCurrentCss As String
    Dim strFragment As String
    Dim strNextCss As String
    Dim strOriginalCell As String
    Dim strResult As String
    Dim strValue As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    blnHasOutputToken = _
        InStr(1, strTemplate, "[OUTPUT_LOCATIONS]", vbTextCompare) > 0 Or _
        InStr(1, strTemplate, "[SHAREPOINT_URL]", vbTextCompare) > 0

    If Not rngTemplate Is Nothing Then
        strOriginalCell = CStr(rngTemplate.Value2)
        If StrComp(Trim$(strOriginalCell), strTemplate, vbBinaryCompare) = 0 Then
            lngCellOffset = InStr(1, strOriginalCell, strTemplate, vbBinaryCompare) - 1
            blnUseRichText = lngCellOffset >= 0
        End If
    End If

    lngPosition = 1
    Do While lngPosition <= Len(strTemplate)
        strValue = vbNullString
        strFragment = vbNullString

        ' A Mapping link is written as [[Link label|https://folder-url]].
        lngConsumed = 0
        If Mid$(strTemplate, lngPosition, 2) = "[[" Then
            lngConsumed = TryRenderInlineMappingLink( _
                strTemplate, lngPosition, strFragment)
        End If

        If lngConsumed = 0 Then
            lngConsumed = 1
            Select Case True
            Case StartsWithMailToken(strTemplate, lngPosition, LEGACY_SHAREPOINT_LINE, True)
                lngConsumed = Len(LEGACY_SHAREPOINT_LINE)
                strFragment = strOutputLocationsHtml
            Case StartsWithMailToken(strTemplate, lngPosition, LEGACY_HEADER, True)
                lngConsumed = Len(LEGACY_HEADER)
                strValue = NEW_HEADER
            Case StartsWithMailToken(strTemplate, lngPosition, "[OUTPUT_LOCATIONS]", True)
                lngConsumed = Len("[OUTPUT_LOCATIONS]")
                strFragment = strOutputLocationsHtml
            Case StartsWithMailToken( _
                strTemplate, lngPosition, "[MASTER_HISTORICAL_FOLDER]", True)
                lngConsumed = Len("[MASTER_HISTORICAL_FOLDER]")
                strFragment = strHistoricalFolderHtml
            Case StartsWithMailToken(strTemplate, lngPosition, "[SHAREPOINT_URL]", True)
                lngConsumed = Len("[SHAREPOINT_URL]")
                strFragment = strOutputLocationsHtml
            Case StartsWithMailToken(strTemplate, lngPosition, "[RUN_DATE]", False)
                lngConsumed = Len("[RUN_DATE]")
                strValue = strRunDate
            Case StartsWithMailToken(strTemplate, lngPosition, "[CHECK_DATE]", False)
                lngConsumed = Len("[CHECK_DATE]")
                strValue = strRunDate
            Case StartsWithMailToken(strTemplate, lngPosition, "[TEAMS]", False)
                lngConsumed = Len("[TEAMS]")
                strValue = strTeams
            Case StartsWithMailToken(strTemplate, lngPosition, "[LOCAL_FOLDER]", False)
                lngConsumed = Len("[LOCAL_FOLDER]")
                strValue = strLocalFolder
            Case StartsWithMailToken(strTemplate, lngPosition, "[TRANSACTION_COUNT]", False)
                lngConsumed = Len("[TRANSACTION_COUNT]")
                strValue = CStr(lngTransactionCount)
            Case StartsWithMailToken(strTemplate, lngPosition, "[UNKNOWN_COUNT]", False)
                lngConsumed = Len("[UNKNOWN_COUNT]")
                strValue = CStr(lngUnknownCount)
            Case StartsWithMailToken(strTemplate, lngPosition, "[UNKNOWN_MASTER_FILE]", False)
                lngConsumed = Len("[UNKNOWN_MASTER_FILE]")
                strValue = HISTORICAL_UNKNOWN_FILE
            Case StartsWithMailToken(strTemplate, lngPosition, "[UNKNOWN_STATUS]", False)
                lngConsumed = Len("[UNKNOWN_STATUS]")
                strValue = strUnknownStatus
            Case Else
                strValue = Mid$(strTemplate, lngPosition, 1)
                If strValue = vbCr Then
                    If Mid$(strTemplate, lngPosition + 1, 1) = vbLf Then lngConsumed = 2
                    strFragment = "<br>"
                ElseIf strValue = vbLf Then
                    strFragment = "<br>"
                End If
            End Select
        End If

        ' Dynamic text is encoded. Only validated links from our helpers use raw HTML.
        If Len(strFragment) = 0 Then strFragment = EncodeMailTextAsHtml(strValue)

        strNextCss = vbNullString
        If blnUseRichText Then
            strNextCss = GetMappingCharacterCss(rngTemplate, lngCellOffset + lngPosition)
        End If

        If StrComp(strNextCss, strCurrentCss, vbBinaryCompare) <> 0 Then
            If Len(strCurrentCss) > 0 Then strResult = strResult & "</span>"
            If Len(strNextCss) > 0 Then
                strResult = strResult & "<span style='" & strNextCss & "'>"
            End If
            strCurrentCss = strNextCss
        End If

        strResult = strResult & strFragment
        lngPosition = lngPosition + lngConsumed
    Loop

    If Len(strCurrentCss) > 0 Then strResult = strResult & "</span>"

    If Not blnHasOutputToken Then
        strResult = strResult & "<br><br>" & NEW_HEADER & "<br>" & strOutputLocationsHtml
    End If

ExitPoint:
    If errNumber = 0 Then RenderMailTemplateHtml = strResult
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "characterPosition", lngPosition)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-29
' Parameters:    strTemplate, lngPosition, strLinkHtml (output)
' Returns:       Number of consumed characters (zero when not a Mapping link).
' Description:   Renders [[label|URL]] as a safe clickable link in Outlook.
'-------------------------------------------------------------------------------
Private Function TryRenderInlineMappingLink( _
    ByVal strTemplate As String, _
    ByVal lngPosition As Long, _
    ByRef strLinkHtml As String) As Long
    Const METHOD_NAME As String = "TryRenderInlineMappingLink"
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngClose As Long
    Dim lngConsumed As Long
    Dim lngSeparator As Long
    Dim strAddress As String
    Dim strLabel As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    If Mid$(strTemplate, lngPosition, 2) = "[[" Then
        lngClose = InStr(lngPosition + 2, strTemplate, "]]", vbBinaryCompare)
        If lngClose > 0 Then
            lngSeparator = InStr(lngPosition + 2, strTemplate, "|", vbBinaryCompare)

            If lngSeparator > lngPosition + 2 And lngSeparator < lngClose Then
                strLabel = Trim$(Mid$( _
                    strTemplate, lngPosition + 2, lngSeparator - lngPosition - 2))
                strAddress = Trim$(Mid$( _
                    strTemplate, lngSeparator + 1, lngClose - lngSeparator - 1))

                If Len(strLabel) > 0 And Len(strAddress) > 0 Then
                    If Left$(strAddress, 2) = "\\" Then
                        strAddress = "file:" & Replace$(strAddress, "\", "/")
                    ElseIf Not ( _
                        LCase$(Left$(strAddress, 8)) = "https://" Or _
                        LCase$(Left$(strAddress, 7)) = "http://" Or _
                        LCase$(Left$(strAddress, 7)) = "file://") Then
                        Call VBA.Err.Raise( _
                            ERROR_EMAIL, METHOD_NAME, _
                            "Invalid Mapping hyperlink. Use an HTTPS URL or UNC folder path.")
                    End If

                    strLinkHtml = "<a href=" & Chr$(34) & HtmlEncode(strAddress) & _
                        Chr$(34) & ">" & HtmlEncode(strLabel) & "</a>"
                    lngConsumed = lngClose + 2 - lngPosition
                End If
            End If
        End If
    End If

ExitPoint:
    If errNumber = 0 Then TryRenderInlineMappingLink = lngConsumed
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "characterPosition", lngPosition)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-29
' Parameters:    strText, lngStart, strToken, blnIgnoreCase
' Returns:       Boolean
' Description:   Checks whether a complete placeholder begins at this position.
'-------------------------------------------------------------------------------
Private Function StartsWithMailToken( _
    ByVal strText As String, _
    ByVal lngStart As Long, _
    ByVal strToken As String, _
    ByVal blnIgnoreCase As Boolean) As Boolean
    Const METHOD_NAME As String = "StartsWithMailToken"
    Dim blnResult As Boolean
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngCompareMode As Long

    If Not DEV_MODE Then On Error GoTo ErrHandler

    lngCompareMode = vbBinaryCompare
    If blnIgnoreCase Then lngCompareMode = vbTextCompare

    If lngStart + Len(strToken) - 1 <= Len(strText) Then
        blnResult = StrComp( _
            Mid$(strText, lngStart, Len(strToken)), _
            strToken, lngCompareMode) = 0
    End If

ExitPoint:
    If errNumber = 0 Then StartsWithMailToken = blnResult
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "position", lngStart)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-29
' Parameters:    strText As String
' Returns:       String
' Description:   Encodes cell text and preserves Excel line breaks in HTML.
'-------------------------------------------------------------------------------
Private Function EncodeMailTextAsHtml(ByVal strText As String) As String
    Const METHOD_NAME As String = "EncodeMailTextAsHtml"
    Dim errDescription As String
    Dim errNumber As Long
    Dim strResult As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    strResult = HtmlEncode(strText)
    strResult = Replace$(strResult, vbCrLf, "<br>")
    strResult = Replace$(strResult, vbCr, "<br>")
    strResult = Replace$(strResult, vbLf, "<br>")

ExitPoint:
    If errNumber = 0 Then EncodeMailTextAsHtml = strResult
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-29
' Parameters:    rngTemplate As Excel.Range; lngPosition As Long
' Returns:       String
' Description:   Converts explicit Excel character font properties into CSS.
'-------------------------------------------------------------------------------
Private Function GetMappingCharacterCss( _
    ByVal rngTemplate As Excel.Range, _
    ByVal lngPosition As Long) As String
    Const METHOD_NAME As String = "GetMappingCharacterCss"
    Dim blnStrike As Boolean
    Dim blnUnderlined As Boolean
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngColor As Long
    Dim objFont As Excel.Font
    Dim strFontName As String
    Dim strResult As String
    Dim strSize As String
    Dim varColor As Variant
    Dim varColorIndex As Variant

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set objFont = rngTemplate.Characters(lngPosition, 1).Font

    strFontName = Trim$(CStr(objFont.Name))
    If Len(strFontName) > 0 Then
        strResult = "font-family:&quot;" & HtmlEncode(strFontName) & "&quot;;"
    End If

    If IsNumeric(objFont.Size) Then
        strSize = Replace$(Format$(CDbl(objFont.Size), "0.0"), ",", ".")
        strResult = strResult & "font-size:" & strSize & "pt;"
    End If

    If CBool(objFont.Bold) Then strResult = strResult & "font-weight:bold;"
    If CBool(objFont.Italic) Then strResult = strResult & "font-style:italic;"

    blnUnderlined = (CLng(objFont.Underline) <> xlUnderlineStyleNone)
    blnStrike = CBool(objFont.Strikethrough)
    If blnUnderlined And blnStrike Then
        strResult = strResult & "text-decoration:underline line-through;"
    ElseIf blnUnderlined Then
        strResult = strResult & "text-decoration:underline;"
    ElseIf blnStrike Then
        strResult = strResult & "text-decoration:line-through;"
    End If

    varColorIndex = objFont.ColorIndex
    If Not IsNull(varColorIndex) Then
        If CLng(varColorIndex) <> xlColorIndexAutomatic Then
            varColor = objFont.Color
            If IsNumeric(varColor) Then
                lngColor = CLng(varColor)
                strResult = strResult & "color:#" & _
                    Right$("0" & Hex$(lngColor And &HFF&), 2) & _
                    Right$("0" & Hex$((lngColor \ &H100&) And &HFF&), 2) & _
                    Right$("0" & Hex$((lngColor \ &H10000) And &HFF&), 2) & ";"
            End If
        End If
    End If

ExitPoint:
    Set objFont = Nothing
    If errNumber = 0 Then GetMappingCharacterCss = strResult
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "characterPosition", lngPosition)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-29
' Parameters:    strConfiguredFolder As String
' Returns:       String
' Description:   Renders a fixed local/UNC historical folder link like OUTPUT_LOCATIONS.
'-------------------------------------------------------------------------------
Private Function BuildMasterHistoricalFolderHtml( _
    ByVal strConfiguredFolder As String) As String
    Const METHOD_NAME As String = "BuildMasterHistoricalFolderHtml"
    Dim errDescription As String
    Dim errNumber As Long
    Dim strFolder As String
    Dim strHref As String
    Dim strResult As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    If Len(Trim$(strConfiguredFolder)) = 0 Then
        Call VBA.Err.Raise( _
            ERROR_EMAIL, METHOD_NAME, _
            "MASTER_HISTORICAL_FOLDER_PATH is required when the mail template " & _
            "contains [MASTER_HISTORICAL_FOLDER].")
    End If

    If IsHttpPath(strConfiguredFolder) Then
        Call VBA.Err.Raise( _
            ERROR_EMAIL, METHOD_NAME, _
            "MASTER_HISTORICAL_FOLDER_PATH must be a local drive or UNC path.")
    End If

    strFolder = ResolveConfiguredPath(strConfiguredFolder)

    If Left$(strFolder, 2) = "\\" Then
        strHref = "file:" & Replace$(strFolder, "\", "/")
    ElseIf Len(strFolder) >= 3 And Mid$(strFolder, 2, 2) = ":\" Then
        strHref = "file:///" & Replace$(strFolder, "\", "/")
    Else
        Call VBA.Err.Raise( _
            ERROR_EMAIL, METHOD_NAME, _
            "MASTER_HISTORICAL_FOLDER_PATH must be a full drive or UNC path.")
    End If

    strResult = "<a href='" & HtmlEncode(strHref) & "'>" & _
        HtmlEncode(strFolder) & "</a>"

ExitPoint:
    If errNumber = 0 Then BuildMasterHistoricalFolderHtml = strResult
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-20
' Parameters:    Configuration plus local and SharePoint output locations.
' Returns:       String
' Description:   Builds the dynamic HTML list of enabled output locations for the email.
'-------------------------------------------------------------------------------
Private Function BuildOutputLocationsHtml( _
    ByRef appConfig As TAppConfig, _
    ByVal strLocalFolder As String, _
    ByVal strSharePointUrl As String, _
    ByVal strSharePointFolder As String) As String
    Const METHOD_NAME As String = "BuildOutputLocationsHtml"
    Dim errDescription As String
    Dim errNumber As Long
    Dim strEncodedLocal As String
    Dim strEncodedSharePoint As String
    Dim strLocalHref As String
    Dim strResult As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    If appConfig.SaveLocal Then
        strEncodedLocal = HtmlEncode(strLocalFolder)
        strLocalHref = Replace$(strLocalFolder, "\", "/")

        If Left$(strLocalHref, 2) = "//" Then
            strLocalHref = "file:" & strLocalHref
        Else
            strLocalHref = "file:///" & strLocalHref
        End If

        strLocalHref = HtmlEncode(strLocalHref)
        strResult = _
            "Local folder: <a href='" & strLocalHref & "'>" & _
            strEncodedLocal & "</a>"
    End If

    If appConfig.SaveSharePoint Then
        If Len(strResult) > 0 Then strResult = strResult & "<br>"

        If Len(Trim$(strSharePointUrl)) > 0 Then
            strEncodedSharePoint = HtmlEncode(strSharePointUrl)
            strResult = _
                strResult & "SharePoint: <a href='" & strEncodedSharePoint & "'>" & _
                strEncodedSharePoint & "</a>"
        Else
            strResult = strResult & "SharePoint folder: " & HtmlEncode(strSharePointFolder)
        End If
    End If

ExitPoint:
    If errNumber = 0 Then BuildOutputLocationsHtml = strResult
    If errNumber <> 0 Then Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "saveLocal;saveSharePoint", appConfig.SaveLocal, appConfig.SaveSharePoint)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-27
' Parameters:    strValue As String
' Returns:       String
' Description:   HTML-encodes text before insertion into Outlook HTMLBody.
'-------------------------------------------------------------------------------
Private Function HtmlEncode(ByVal strValue As String) As String
    Const METHOD_NAME As String = "HtmlEncode"
    Dim errDescription As String
    Dim errNumber As Long
    Dim strResult As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    strResult = strValue
    strResult = Replace$(strResult, "&", "&amp;")
    strResult = Replace$(strResult, "<", "&lt;")
    strResult = Replace$(strResult, ">", "&gt;")
    strResult = Replace$(strResult, Chr$(34), "&quot;")
    strResult = Replace$(strResult, "'", "&#39;")

ExitPoint:
    If errNumber = 0 Then HtmlEncode = strResult
    If errNumber <> 0 Then Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Function

' Author:        Pawel Ligezka
' Creation date: 2026-08-27
' Parameters:    objMail As Object
' Returns:       Boolean
' Description:   Applies the Internal MIP label on a best-effort basis without blocking draft creation.
'-------------------------------------------------------------------------------
Private Function TryApplyInternalLabel(ByVal objMail As Object) As Boolean
    Const METHOD_NAME As String = "TryApplyInternalLabel"
    Dim blnResult As Boolean
    Dim errDescription As String
    Dim errNumber As Long

    If Not DEV_MODE Then On Error GoTo ErrHandler

    If APPLY_INTERNAL_LABEL Then
        objMail.PropertyAccessor.SetProperty MSIP_LABEL_SCHEMA, INTERNAL_LABEL_METADATA
    End If

    blnResult = True

ExitPoint:
    TryApplyInternalLabel = blnResult
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription)

    ' This is an explicitly best-effort operation. The draft must still be
    ' created when the Outlook label cannot be applied, so the error is logged
    ' and converted to a False return value instead of being propagated.
    blnResult = False
    GoTo ExitPoint
End Function
