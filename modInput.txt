Option Explicit

Private Const CLASS_NAME As String = "modInput"
Private Const AD_SCHEMA_TABLES As Long = 20
Private Const AD_OPEN_FORWARD_ONLY As Long = 0
Private Const AD_LOCK_READ_ONLY As Long = 1

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-27
' Parameters:    Attachment paths; worksheet option(s); attachment subjects
' Returns:       Variant
' Description:   Merges source files after mapping input columns by header name.
'-------------------------------------------------------------------------------
Public Function MergeInputFiles( _
    ByVal colAttachmentPaths As Collection, _
    ByVal strWorksheetName As String, _
    ByVal dictAttachmentSubjects As Object) As Variant
    Const METHOD_NAME As String = "MergeInputFiles"
    Dim arrCombined As Variant
    Dim arrFileData As Variant
    Dim arrWorking As Variant
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngAttachmentCount As Long
    Dim lngCapacity As Long
    Dim lngColumn As Long
    Dim lngFile As Long
    Dim lngFileRows As Long
    Dim lngRecord As Long
    Dim lngRecordCount As Long
    Dim lngRequiredCapacity As Long
    Dim lngRow As Long
    Dim strFilePath As String
    Dim strMailSubject As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    If colAttachmentPaths Is Nothing Then
        Call VBA.Err.Raise( _
            ERROR_INPUT_DATA, METHOD_NAME, _
            "Attachment collection is not available.")
    End If

    lngAttachmentCount = colAttachmentPaths.Count
    If lngAttachmentCount = 0 Then
        Call VBA.Err.Raise( _
            ERROR_INPUT_DATA, METHOD_NAME, _
            "No input files were supplied for merging.")
    End If

    lngCapacity = INITIAL_INPUT_CAPACITY
    ReDim arrWorking(1 To OUTPUT_COLUMN_COUNT, 1 To lngCapacity)

    For lngFile = 1 To lngAttachmentCount
        strFilePath = CStr(colAttachmentPaths(lngFile))
        strMailSubject = vbNullString

        If Not dictAttachmentSubjects Is Nothing Then
            If dictAttachmentSubjects.Exists(strFilePath) Then
                strMailSubject = CStr(dictAttachmentSubjects(strFilePath))
            End If
        End If

        Application.StatusBar = _
            "Dallas Cash Transactions: merging input file " & _
            CStr(lngFile) & "/" & CStr(lngAttachmentCount) & " - " & _
            GetFileNameFromPath(strFilePath)
        DoEvents
        Call ReadInputFileData(strFilePath, strWorksheetName, strMailSubject, arrFileData, lngFileRows)

        If lngFileRows > 0 Then
            lngRequiredCapacity = lngRecordCount + lngFileRows
            Call EnsureWorkingCapacity(arrWorking, lngCapacity, lngRequiredCapacity)

            For lngRow = 1 To lngFileRows
                lngRecordCount = lngRecordCount + 1

                For lngColumn = 1 To INPUT_COLUMN_COUNT
                    arrWorking(lngColumn + 5, lngRecordCount) = arrFileData(lngRow, lngColumn)
                Next lngColumn
            Next lngRow
        End If

        arrFileData = Empty
    Next lngFile

    ReDim arrCombined(1 To lngRecordCount + 1, 1 To OUTPUT_COLUMN_COUNT)
    Call SetOutputHeaders(arrCombined)

    For lngRecord = 1 To lngRecordCount
        For lngColumn = 6 To OUTPUT_COLUMN_COUNT
            arrCombined(lngRecord + 1, lngColumn) = arrWorking(lngColumn, lngRecord)
        Next lngColumn
    Next lngRecord

ExitPoint:
    If errNumber = 0 Then MergeInputFiles = arrCombined
    If errNumber <> 0 Then Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "attachmentCount;currentFile;worksheetOptions", _
        lngAttachmentCount, strFilePath, strWorksheetName)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-27
' Parameters:    strFilePath As String
' Returns:       Object
' Description:   Opens an ACE OLE DB connection to an Excel input without opening Excel UI.
'-------------------------------------------------------------------------------
Private Function OpenInputDataConnection(ByVal strFilePath As String) As Object
    Const METHOD_NAME As String = "OpenInputDataConnection"
    Dim errDescription As String
    Dim errNumber As Long
    Dim objConnection As Object
    Dim strConnection As String
    Dim strExtension As String
    Dim strExtendedProperties As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    strExtension = LCase$(GetFileExtension(strFilePath))

    Select Case strExtension
        Case ".xlsx"
            strExtendedProperties = "Excel 12.0 Xml;HDR=NO;IMEX=1"
        Case ".xls"
            strExtendedProperties = "Excel 8.0;HDR=NO;IMEX=1"
        Case Else
            Call VBA.Err.Raise( _
                ERROR_INPUT_DATA, METHOD_NAME, _
                "Unsupported input extension for data-only reader: " & strExtension)
    End Select

    strConnection = _
        "Provider=Microsoft.ACE.OLEDB.12.0;Data Source=" & strFilePath & _
        ";Mode=Read;Extended Properties=""" & strExtendedProperties & """;"

    Set objConnection = CreateObject("ADODB.Connection")
    Call objConnection.Open(strConnection)

ExitPoint:
    If errNumber = 0 Then Set OpenInputDataConnection = objConnection
    Set objConnection = Nothing

    If errNumber <> 0 Then
        Call VBA.Err.Raise( _
            errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    errDescription = _
        "The input file could not be read through the ACE OLE DB data reader. " & _
        "This reader avoids Excel/Outlook temporary-file links. Original error: " & _
        errDescription
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "filePath", strFilePath)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-27
' Parameters:    objConnection As Object; strWorksheetOptions As String
' Returns:       String
' Description:   Resolves the first configured worksheet available through ACE schema.
'-------------------------------------------------------------------------------
Private Function ResolveAdoWorksheetName( _
    ByVal objConnection As Object, _
    ByVal strWorksheetOptions As String) As String

    Const METHOD_NAME As String = "ResolveAdoWorksheetName"
    Dim arrOptions As Variant
    Dim dictSheets As Object
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngOption As Long
    Dim objSchema As Object
    Dim strNormalized As String
    Dim strOption As String
    Dim strRawName As String
    Dim strResult As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set dictSheets = CreateObject("Scripting.Dictionary")
    dictSheets.CompareMode = vbTextCompare
    Set objSchema = objConnection.OpenSchema(AD_SCHEMA_TABLES)

    Do While Not objSchema.EOF
        strRawName = CStr(objSchema.Fields("TABLE_NAME").Value)
        strNormalized = NormalizeAdoWorksheetName(strRawName)

        If Len(strNormalized) > 0 Then
            If Not dictSheets.Exists(strNormalized) Then
                dictSheets.Add strNormalized, GetAdoQueryTableName(strRawName)
            End If
        End If

        Call objSchema.MoveNext
    Loop

    arrOptions = Split(strWorksheetOptions, ";")

    For lngOption = LBound(arrOptions) To UBound(arrOptions)
        strOption = Trim$(CStr(arrOptions(lngOption)))

        If Len(strOption) > 0 Then
            If dictSheets.Exists(strOption) Then
                strResult = CStr(dictSheets(strOption))
                Exit For
            End If
        End If
    Next lngOption

    If Len(strResult) = 0 Then
        Call VBA.Err.Raise( _
            ERROR_INPUT_DATA, METHOD_NAME, _
            "None of the configured input worksheets was found: " & _
            strWorksheetOptions)
    End If

ExitPoint:
    Set objSchema = Nothing
    Set dictSheets = Nothing
    If errNumber = 0 Then ResolveAdoWorksheetName = strResult

    If errNumber <> 0 Then
        Call VBA.Err.Raise( _
            errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "worksheetOptions", strWorksheetOptions)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-27
' Parameters:    strRawName As String
' Returns:       String
' Description:   Normalizes an ACE schema table name to a worksheet name.
'-------------------------------------------------------------------------------
Private Function NormalizeAdoWorksheetName(ByVal strRawName As String) As String
    Const METHOD_NAME As String = "NormalizeAdoWorksheetName"
    Dim errDescription As String
    Dim errNumber As Long
    Dim strResult As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    strResult = Trim$(strRawName)

    If Len(strResult) >= 2 Then
        If Left$(strResult, 1) = "'" And Right$(strResult, 1) = "'" Then
            strResult = Mid$(strResult, 2, Len(strResult) - 2)
        End If
    End If

    If Right$(strResult, 1) = "$" Then
        strResult = Left$(strResult, Len(strResult) - 1)
    Else
        strResult = vbNullString
    End If

ExitPoint:
    If errNumber = 0 Then NormalizeAdoWorksheetName = strResult
    If errNumber <> 0 Then
        Call VBA.Err.Raise( _
            errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "rawName", strRawName)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-27
' Parameters:    objConnection As Object; strQueryTableName As String
' Returns:       Variant
' Description:   Reads a worksheet into a row-major array without opening an Excel workbook.
'-------------------------------------------------------------------------------
Private Function ReadAdoWorksheet( _
    ByVal objConnection As Object, _
    ByVal strQueryTableName As String) As Variant

    Const METHOD_NAME As String = "ReadAdoWorksheet"
    Dim arrRaw As Variant
    Dim arrResult As Variant
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngColumn As Long
    Dim lngColumnCount As Long
    Dim lngRow As Long
    Dim lngRowCount As Long
    Dim objRecordset As Object
    Dim strSql As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    strSql = "SELECT * FROM [" & strQueryTableName & "]"
    Set objRecordset = CreateObject("ADODB.Recordset")
    Call objRecordset.Open( _
        strSql, objConnection, AD_OPEN_FORWARD_ONLY, AD_LOCK_READ_ONLY)

    If Not (objRecordset.BOF And objRecordset.EOF) Then
        lngColumnCount = objRecordset.Fields.Count
        arrRaw = objRecordset.GetRows()
        lngRowCount = UBound(arrRaw, 2) + 1
        ReDim arrResult(1 To lngRowCount, 1 To lngColumnCount)

        For lngRow = 1 To lngRowCount
            For lngColumn = 1 To lngColumnCount
                arrResult(lngRow, lngColumn) = _
                    arrRaw(lngColumn - 1, lngRow - 1)
            Next lngColumn
        Next lngRow
    End If

ExitPoint:
    Set objRecordset = Nothing
    If errNumber = 0 Then ReadAdoWorksheet = arrResult

    If errNumber <> 0 Then
        Call VBA.Err.Raise( _
            errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "worksheet", strQueryTableName)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-27
' Parameters:    arrSheet As Variant
' Returns:       Long
' Description:   Finds the last populated header column in an ADO worksheet array.
'-------------------------------------------------------------------------------
Private Function GetAdoLastHeaderColumn(ByRef arrSheet As Variant) As Long
    Const METHOD_NAME As String = "GetAdoLastHeaderColumn"
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngColumn As Long
    Dim lngResult As Long

    If Not DEV_MODE Then On Error GoTo ErrHandler

    If IsArray(arrSheet) Then
        For lngColumn = UBound(arrSheet, 2) To LBound(arrSheet, 2) Step -1
            If Len(GetHeaderText(arrSheet(1, lngColumn))) > 0 Then
                lngResult = lngColumn
                Exit For
            End If
        Next lngColumn
    End If

ExitPoint:
    If errNumber = 0 Then GetAdoLastHeaderColumn = lngResult
    If errNumber <> 0 Then
        Call VBA.Err.Raise( _
            errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "column", lngColumn)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-27
' Parameters:    File path, worksheet options, mail subject, output data and row count.
' Returns:       ---
' Description:   Reads one input through ACE OLE DB and normalizes columns by header.
'-------------------------------------------------------------------------------
Private Sub ReadInputFileData( _
    ByVal strFilePath As String, _
    ByVal strWorksheetName As String, _
    ByVal strMailSubject As String, _
    ByRef arrData As Variant, _
    ByRef lngDataRows As Long)

    Const METHOD_NAME As String = "ReadInputFileData"
    Dim arrSheet As Variant
    Dim arrSource As Variant
    Dim cleanupErrNumber As Long
    Dim dictColumnMap As Object
    Dim errDescription As String
    Dim errNumber As Long
    Dim handlerErrDescription As String
    Dim handlerErrNumber As Long
    Dim lngLastColumn As Long
    Dim lngOutputRow As Long
    Dim lngSheetRowCount As Long
    Dim lngSourceColumn As Long
    Dim lngSourceRow As Long
    Dim lngTargetColumn As Long
    Dim objConnection As Object
    Dim strAccountContext As String
    Dim strDisplayedAccount As String
    Dim strQuerySheet As String
    Dim varValue As Variant

    If Not DEV_MODE Then On Error GoTo ErrHandler

    arrData = Empty
    lngDataRows = 0
    Set objConnection = OpenInputDataConnection(strFilePath)
    strQuerySheet = ResolveAdoWorksheetName( _
        objConnection, strWorksheetName)
    arrSheet = ReadAdoWorksheet(objConnection, strQuerySheet)

    If IsArray(arrSheet) Then
        lngSheetRowCount = UBound(arrSheet, 1)
        lngLastColumn = GetAdoLastHeaderColumn(arrSheet)

        If lngLastColumn > 0 And lngSheetRowCount >= INPUT_FIRST_DATA_ROW Then
            Set dictColumnMap = BuildInputColumnMap( _
                arrSheet, lngLastColumn, strFilePath, strMailSubject)

            ReDim arrSource( _
                1 To lngSheetRowCount - INPUT_HEADER_ROW, _
                1 To lngLastColumn)

            For lngSourceRow = INPUT_FIRST_DATA_ROW To lngSheetRowCount
                For lngSourceColumn = 1 To lngLastColumn
                    arrSource(lngSourceRow - INPUT_HEADER_ROW, lngSourceColumn) = _
                        arrSheet(lngSourceRow, lngSourceColumn)
                Next lngSourceColumn
            Next lngSourceRow
        End If
    End If

    If IsArray(arrSource) Then
        If ArrayContainsInputData(arrSource) Then
            lngDataRows = CountInputDataRows(arrSource, dictColumnMap)

            If lngDataRows > 0 Then
                ReDim arrData(1 To lngDataRows, 1 To INPUT_COLUMN_COUNT)

                For lngSourceRow = 1 To UBound(arrSource, 1)
                If InputRowHasData(arrSource, lngSourceRow, dictColumnMap) Then
                    lngOutputRow = lngOutputRow + 1

                    For lngTargetColumn = 1 To INPUT_COLUMN_COUNT
                        If dictColumnMap.Exists(lngTargetColumn) Then
                            lngSourceColumn = CLng(dictColumnMap(lngTargetColumn))
                            varValue = arrSource(lngSourceRow, lngSourceColumn)

                            If lngTargetColumn = 1 Then
                                strAccountContext = _
                                    "Input file: '" & strFilePath & "'."

                                If IsError(varValue) Or IsNull(varValue) Or _
                                    IsEmpty(varValue) Then

                                    strDisplayedAccount = vbNullString
                                Else
                                    strDisplayedAccount = Trim$(CStr(varValue))
                                End If

                                arrData(lngOutputRow, lngTargetColumn) = _
                                    NormalizeAccountIdentifier( _
                                        varValue, strDisplayedAccount, _
                                        ERROR_INPUT_DATA, strAccountContext)
                            ElseIf lngTargetColumn = 3 Or _
                                lngTargetColumn = 4 Then

                                strAccountContext = _
                                    "Input file: '" & strFilePath & "'."
                                arrData(lngOutputRow, lngTargetColumn) = _
                                    NormalizeInputDateValue( _
                                        varValue, strAccountContext)
                            Else
                                arrData(lngOutputRow, lngTargetColumn) = varValue
                            End If
                        Else
                            arrData(lngOutputRow, lngTargetColumn) = vbNullString
                        End If
                    Next lngTargetColumn
                End If
                Next lngSourceRow
            End If
        End If
    End If

ExitPoint:
    Set dictColumnMap = Nothing

    If Not objConnection Is Nothing Then
        cleanupErrNumber = CloseAdoConnectionSafely(objConnection)

        If errNumber = 0 And cleanupErrNumber <> 0 Then
            errNumber = cleanupErrNumber
            errDescription = _
                "The input data connection could not be closed during cleanup."
        End If
    End If

    Set objConnection = Nothing

    If errNumber <> 0 Then
        Call VBA.Err.Raise( _
            errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Sub

ErrHandler:
    handlerErrNumber = VBA.Err.Number
    handlerErrDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, handlerErrNumber, handlerErrDescription, _
        "filePath;worksheetOptions", strFilePath, strWorksheetName)

    If errNumber = 0 Then
        errNumber = handlerErrNumber
        errDescription = handlerErrDescription
    End If

    GoTo ExitPoint
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-27
' Parameters:    varValue As Variant; strContext As String
' Returns:       Variant
' Description:   Normalizes an input date to an Excel date serial without time.
'-------------------------------------------------------------------------------
Private Function NormalizeInputDateValue( _
    ByVal varValue As Variant, _
    ByVal strContext As String) As Variant

    Const METHOD_NAME As String = "NormalizeInputDateValue"
    Dim dblResult As Double
    Dim dtValue As Date
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngDay As Long
    Dim lngMonth As Long
    Dim lngYear As Long
    Dim strValue As String
    Dim varResult As Variant

    If Not DEV_MODE Then On Error GoTo ErrHandler

    If IsError(varValue) Then
        Call VBA.Err.Raise( _
            ERROR_INPUT_DATA, METHOD_NAME, _
            "An Excel error value cannot be converted to a date. " & strContext)
    End If

    If IsNull(varValue) Or IsEmpty(varValue) Then
        varResult = vbNullString
    Else
        strValue = Trim$(CStr(varValue))

        If Len(strValue) = 0 Then
            varResult = vbNullString
        ElseIf IsNumeric(varValue) Then
            dblResult = Fix(CDbl(varValue))

            If dblResult <= 0 Or dblResult > 2958465# Then
                Call VBA.Err.Raise( _
                    ERROR_INPUT_DATA, METHOD_NAME, _
                    "Invalid Excel serial date value. " & strContext)
            End If

            varResult = dblResult
        ElseIf VarType(varValue) = vbDate Then
            dtValue = CDate(varValue)
            varResult = Fix(CDbl(dtValue))
        ElseIf Len(strValue) >= 10 And _
            Mid$(strValue, 5, 1) = "-" And Mid$(strValue, 8, 1) = "-" Then

            If Not IsNumeric(Left$(strValue, 4)) Or _
                Not IsNumeric(Mid$(strValue, 6, 2)) Or _
                Not IsNumeric(Right$(strValue, 2)) Then
                Call VBA.Err.Raise( _
                    ERROR_INPUT_DATA, METHOD_NAME, _
                    "Invalid ISO date value. " & strContext)
            End If

            lngYear = CLng(Left$(strValue, 4))
            lngMonth = CLng(Mid$(strValue, 6, 2))
            lngDay = CLng(Mid$(strValue, 9, 2))
            dtValue = DateSerial(lngYear, lngMonth, lngDay)

            If Year(dtValue) <> lngYear Or _
                Month(dtValue) <> lngMonth Or Day(dtValue) <> lngDay Then
                Call VBA.Err.Raise( _
                    ERROR_INPUT_DATA, METHOD_NAME, _
                    "Invalid ISO date value. " & strContext)
            End If

            varResult = CDbl(dtValue)
        ElseIf IsDate(varValue) Then
            varResult = CDbl(DateValue(CDate(varValue)))
        Else
            Call VBA.Err.Raise( _
                ERROR_INPUT_DATA, METHOD_NAME, _
                "Unsupported input date value. " & strContext)
        End If
    End If

ExitPoint:
    If errNumber = 0 Then NormalizeInputDateValue = varResult
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "context", strContext)
    GoTo ExitPoint
End Function
'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-27
' Parameters:    Header array; last column; source path; source mail subject
' Returns:       Object
' Description:   Maps ADO header positions to the canonical sixteen columns.
'-------------------------------------------------------------------------------
Private Function BuildInputColumnMap( _
    ByRef arrSheet As Variant, _
    ByVal lngLastColumn As Long, _
    ByVal strFilePath As String, _
    ByVal strMailSubject As String) As Object
    Const METHOD_NAME As String = "BuildInputColumnMap"
    Dim arrHeaders As Variant
    Dim dictColumnMap As Object
    Dim dictExpected As Object
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngColumn As Long
    Dim lngTargetColumn As Long
    Dim strActualHeader As String
    Dim strExpectedHeader As String
    Dim strHeaderKey As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    arrHeaders = GetInputHeaders()
    Set dictExpected = CreateObject("Scripting.Dictionary")
    dictExpected.CompareMode = vbTextCompare
    Set dictColumnMap = CreateObject("Scripting.Dictionary")
    dictColumnMap.CompareMode = vbTextCompare

    For lngTargetColumn = 1 To INPUT_COLUMN_COUNT
        strExpectedHeader = CStr(arrHeaders(lngTargetColumn - 1))
        strHeaderKey = NormalizeHeaderName(strExpectedHeader)
        dictExpected(strHeaderKey) = lngTargetColumn
    Next lngTargetColumn

    For lngColumn = 1 To lngLastColumn
        strActualHeader = GetHeaderText(arrSheet(INPUT_HEADER_ROW, lngColumn))
        strHeaderKey = NormalizeHeaderName(strActualHeader)

        If Len(strHeaderKey) > 0 Then
            If dictExpected.Exists(strHeaderKey) Then
                lngTargetColumn = CLng(dictExpected(strHeaderKey))

                If dictColumnMap.Exists(lngTargetColumn) Then
                    strExpectedHeader = CStr(arrHeaders(lngTargetColumn - 1))
                    Call VBA.Err.Raise( _
                        ERROR_INPUT_DATA, METHOD_NAME, _
                        "Duplicate input header matching '" & strExpectedHeader & _
                        "' was found. Input file: '" & strFilePath & "'.")
                End If

                dictColumnMap.Add lngTargetColumn, lngColumn
            End If
        End If
    Next lngColumn

    For lngTargetColumn = 1 To INPUT_COLUMN_COUNT
        If IsRequiredInputHeader(lngTargetColumn) Then
            If Not dictColumnMap.Exists(lngTargetColumn) Then
                strExpectedHeader = CStr(arrHeaders(lngTargetColumn - 1))
                Call VBA.Err.Raise( _
                    ERROR_INPUT_DATA, METHOD_NAME, _
                    "Required input header '" & strExpectedHeader & _
                    "' was not found in a non-empty input file. Input file: '" & _
                    strFilePath & "'.")
            End If
        End If
    Next lngTargetColumn

ExitPoint:
    Set dictExpected = Nothing
    If errNumber = 0 Then Set BuildInputColumnMap = dictColumnMap
    Set dictColumnMap = Nothing
    If errNumber <> 0 Then
        Call VBA.Err.Raise( _

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-27
' Parameters:    strHeader As String
' Returns:       String
' Description:   Normalizes source headers and supported known aliases.
'-------------------------------------------------------------------------------
Private Function NormalizeHeaderName(ByVal strHeader As String) As String
    Const METHOD_NAME As String = "NormalizeHeaderName"
    Dim errDescription As String
    Dim errNumber As Long
    Dim strResult As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    strResult = Trim$(strHeader)
    strResult = Replace$(strResult, Chr$(160), " ")
    strResult = Replace$(strResult, vbCr, " ")
    strResult = Replace$(strResult, vbLf, " ")
    strResult = Replace$(strResult, vbTab, " ")

    Do While InStr(1, strResult, "  ", vbBinaryCompare) > 0
        strResult = Replace$(strResult, "  ", " ")
    Loop

    strResult = LCase$(strResult)
    strResult = Replace$(strResult, ".", vbNullString)

    If StrComp(strResult, "accout name", vbTextCompare) = 0 Then strResult = "account name"

ExitPoint:
    If errNumber = 0 Then NormalizeHeaderName = strResult
    If errNumber <> 0 Then Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError(CLASS_NAME, METHOD_NAME, errNumber, errDescription, "header", strHeader)
    GoTo ExitPoint
End Function

    Const METHOD_NAME As String = "ArrayContainsInputData"

    ByRef arrSource As Variant, _

    ByVal dictColumnMap As Object) As Boolean

    Const METHOD_NAME As String = "InputRowHasData"
    Dim blnResult As Boolean

                blnResult = True
                Exit For
            End If
        End If
    Next lngTargetColumn

ExitPoint:
    If errNumber = 0 Then InputRowHasData = blnResult
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "row;targetColumn", lngRow, lngTargetColumn)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-26
' Parameters:    arrWorking As Variant; lngCapacity As Long; lngRequired As Long
' Returns:       ---
' Description:   Expands the column-major working array using capacity doubling.
'-------------------------------------------------------------------------------
Private Sub EnsureWorkingCapacity( _
    ByRef arrWorking As Variant, _
    ByRef lngCapacity As Long, _
    ByVal lngRequired As Long)

    Const METHOD_NAME As String = "EnsureWorkingCapacity"
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngNewCapacity As Long

    If Not DEV_MODE Then On Error GoTo ErrHandler

    If lngRequired > lngCapacity Then
        lngNewCapacity = lngCapacity

        Do While lngNewCapacity < lngRequired
            lngNewCapacity = lngNewCapacity * 2
        Loop

        ReDim Preserve arrWorking(1 To OUTPUT_COLUMN_COUNT, 1 To lngNewCapacity)
        lngCapacity = lngNewCapacity
    End If

ExitPoint:
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "currentCapacity;requiredCapacity", lngCapacity, lngRequired)
    GoTo ExitPoint
End Sub
