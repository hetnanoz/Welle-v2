Option Explicit

Private Const CLASS_NAME As String = "modFondsliste"

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-26
' Parameters:    appConfig As TAppConfig
' Returns:       String - full path to the one configured Fondsliste file
' Description:   Finds only the exact configured base name and extension.
'                Keeps the original public name for modMain compatibility.
'-------------------------------------------------------------------------------
Public Function GetLatestFondslistePath(ByRef appConfig As TAppConfig) As String
    Const METHOD_NAME As String = "GetLatestFondslistePath"
    Dim dtStepStarted As Date
    Dim errDescription As String
    Dim errNumber As Long
    Dim strExpectedFile As String
    Dim strFolderPath As String
    Dim strFoundFile As String
    Dim strResult As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    dtStepStarted = Now
    strFolderPath = ResolveConfiguredPath(appConfig.FondslisteFolder)
    strExpectedFile = BuildExactFondslisteFileName( _
        appConfig.FondslistePattern, appConfig.FondslisteExtension)

    Debug.Print "Dallas PERF | Fondsliste FIXED expected | " & strExpectedFile
    Debug.Print "Dallas PERF | Fondsliste resolve settings | " & _
        CStr(DateDiff("s", dtStepStarted, Now)) & " s"

    ' Dir$ uses an exact full file path, without a wildcard or a folder scan.
    dtStepStarted = Now
    strResult = CombinePath(strFolderPath, strExpectedFile)
    strFoundFile = Dir$( _
        strResult, vbNormal Or vbReadOnly Or vbHidden Or vbSystem Or vbArchive)

    Debug.Print "Dallas PERF | Fondsliste FIXED file lookup | " & _
        CStr(DateDiff("s", dtStepStarted, Now)) & " s"

    ' Check Dir$ output as well; no short-name/wildcard-like match is accepted.
    If Len(strFoundFile) = 0 Or _
        StrComp(strFoundFile, strExpectedFile, vbTextCompare) <> 0 Then
        Call VBA.Err.Raise( _
            ERROR_FONDSLISTE, METHOD_NAME, _
            "Required FIXED Fondsliste file not found: " & strExpectedFile & _
            ". Folder: " & strFolderPath & ". No other file will be used.")
    End If

    strResult = CombinePath(strFolderPath, strFoundFile)

ExitPoint:
    If errNumber = 0 Then GetLatestFondslistePath = strResult
    If errNumber <> 0 Then
        Call VBA.Err.Raise( _
            errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "folder;baseName;extension", _
        appConfig.FondslisteFolder, appConfig.FondslistePattern, _
        appConfig.FondslisteExtension)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-29
' Parameters:    strFileName - exact base name, without extension
'                strFileExtension - exact extension, with or without leading dot
' Returns:       String - exact file name, including extension
' Description:   Rejects wildcard, path characters and duplicate extension.
'-------------------------------------------------------------------------------
Private Function BuildExactFondslisteFileName( _
    ByVal strFileName As String, _
    ByVal strFileExtension As String) As String
    Const METHOD_NAME As String = "BuildExactFondslisteFileName"
    Const INVALID_FILE_NAME_CHARS As String = "\/:*?""<>|"
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngIndex As Long
    Dim strBaseName As String
    Dim strCharacter As String
    Dim strExtension As String
    Dim strResult As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    strBaseName = Trim$(strFileName)
    strExtension = Trim$(strFileExtension)

    If Len(strBaseName) = 0 Then
        Call VBA.Err.Raise( _
            ERROR_CONFIGURATION, METHOD_NAME, _
            "Fondsliste FIXED base file name cannot be blank.")
    End If
    If Len(strExtension) = 0 Then
        Call VBA.Err.Raise( _
            ERROR_CONFIGURATION, METHOD_NAME, _
            "FONDSLISTE_EXTENSION cannot be blank.")
    End If
    If Left$(strExtension, 1) <> "." Then strExtension = "." & strExtension

    For lngIndex = 1 To Len(INVALID_FILE_NAME_CHARS)
        strCharacter = Mid$(INVALID_FILE_NAME_CHARS, lngIndex, 1)
        If InStr(1, strBaseName, strCharacter, vbBinaryCompare) > 0 Or _
           InStr(1, strExtension, strCharacter, vbBinaryCompare) > 0 Then
            Call VBA.Err.Raise( _
                ERROR_CONFIGURATION, METHOD_NAME, _
                "FIXED Fondsliste name/extension contains an invalid character: " & _
                strCharacter)
        End If
    Next lngIndex

    If strBaseName = "." Or strBaseName = ".." Or _
        Right$(strBaseName, 1) = "." Or _
        InStr(2, strExtension, ".", vbBinaryCompare) > 0 Then
        Call VBA.Err.Raise( _
            ERROR_CONFIGURATION, METHOD_NAME, _
            "Use the base file name without extension and one extension, e.g. .xls.")
    End If

    If EndsWithText(strBaseName, strExtension) Then
        Call VBA.Err.Raise( _
            ERROR_CONFIGURATION, METHOD_NAME, _
            "Fondsliste base name must not include FONDSLISTE_EXTENSION.")
    End If

    strResult = strBaseName & strExtension

ExitPoint:
    If errNumber = 0 Then BuildExactFondslisteFileName = strResult
    If errNumber <> 0 Then
        Call VBA.Err.Raise( _
            errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "baseName;extension", strFileName, strFileExtension)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-26
' Parameters:    strValue As String; strSuffix As String
' Returns:       Boolean
' Description:   Checks whether text ends with a suffix, ignoring case.
'-------------------------------------------------------------------------------
Private Function EndsWithText(ByVal strValue As String, ByVal strSuffix As String) As Boolean
    Const METHOD_NAME As String = "EndsWithText"
    Dim blnResult As Boolean
    Dim errDescription As String
    Dim errNumber As Long

    If Not DEV_MODE Then On Error GoTo ErrHandler

    If Len(strSuffix) <= Len(strValue) Then
        blnResult = _
            StrComp( _
                Right$(strValue, Len(strSuffix)), strSuffix, vbTextCompare) = 0
    End If

ExitPoint:
    If errNumber = 0 Then EndsWithText = blnResult
    If errNumber <> 0 Then Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-26
' Parameters:    strFondslistePath; blnUseLocalCopy; strWorkspace
' Returns:       Object
' Description:   Builds one Open Fund Number to Team dictionary in memory.
'                Optional local staging avoids Excel opening remote files directly.
'-------------------------------------------------------------------------------
Public Function BuildOpenFundTeamDictionary( _
    ByVal strFondslistePath As String, _
    ByVal blnUseLocalCopy As Boolean, _
    ByVal strWorkspace As String) As Object

    Const METHOD_NAME As String = "BuildOpenFundTeamDictionary"
    Dim arrData As Variant
    Dim blnCloseRequired As Boolean
    Dim cleanupErrNumber As Long
    Dim dictTeams As Object
    Dim dtStepStarted As Date
    Dim errDescription As String
    Dim errNumber As Long
    Dim handlerErrDescription As String
    Dim handlerErrNumber As Long
    Dim lngCopyErrNumber As Long
    Dim strCurrentOperation As String
    Dim lngLastRow As Long
    Dim lngRow As Long
    Dim strFundNumber As String
    Dim strOpenPath As String
    Dim strStagedCopyPath As String
    Dim strStatus As String
    Dim strTeam As String
    Dim wkbAlreadyOpen As Excel.Workbook
    Dim wkbFonds As Excel.Workbook
    Dim wksFonds As Excel.Worksheet

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set dictTeams = CreateObject("Scripting.Dictionary")
    dictTeams.CompareMode = vbTextCompare
    strOpenPath = strFondslistePath

    strCurrentOperation = "find already-open source workbook"
    dtStepStarted = Now
    Set wkbAlreadyOpen = GetOpenWorkbookByFullName(strFondslistePath, True)
    Debug.Print "Dallas PERF | Fondsliste existing-workbook lookup | " & _
        CStr(DateDiff("s", dtStepStarted, Now)) & " s"

    If wkbAlreadyOpen Is Nothing Then
        If blnUseLocalCopy Then
            If Len(Trim$(strWorkspace)) = 0 Then
                Call VBA.Err.Raise( _
                    ERROR_FONDSLISTE, METHOD_NAME, _
                    "Temporary workspace is required for Fondsliste local copy.")
            End If

            strStagedCopyPath = CombinePath( _
                strWorkspace, "DallasFondsliste_" & GetFileNameFromPath(strFondslistePath))
            strCurrentOperation = "copy Fondsliste to temporary workspace"
            Debug.Print "Dallas PERF | Fondsliste local copy | starting"
            dtStepStarted = Now

            If TryStageFondslisteCopy( _
                strFondslistePath, strStagedCopyPath, lngCopyErrNumber) Then
                strOpenPath = strStagedCopyPath
                Debug.Print "Dallas PERF | Fondsliste network-to-local copy | " & _
                    CStr(DateDiff("s", dtStepStarted, Now)) & " s"
            Else
                Debug.Print "Dallas PERF | Fondsliste local copy failed (error " & _
                    CStr(lngCopyErrNumber) & ") | " & _
                    CStr(DateDiff("s", dtStepStarted, Now)) & " s"
                Debug.Print "Dallas PERF | Fondsliste fallback | direct read-only opening"
                ' The optional performance optimization must not block the run.
                ' strOpenPath still points to the exact selected source file.
            End If
        End If

        strCurrentOperation = "open Fondsliste workbook"
        Debug.Print "Dallas PERF | Fondsliste Workbooks.Open | starting"
        dtStepStarted = Now
        Set wkbFonds = Application.Workbooks.Open( _
            Filename:=strOpenPath, UpdateLinks:=0, ReadOnly:=True, _
            IgnoreReadOnlyRecommended:=True, AddToMru:=False, Notify:=False)
        blnCloseRequired = True
        Debug.Print "Dallas PERF | Fondsliste Workbooks.Open | " & _
            CStr(DateDiff("s", dtStepStarted, Now)) & " s"
    Else
        Set wkbFonds = wkbAlreadyOpen
        Debug.Print "Dallas PERF | Fondsliste already open: reused workbook"
    End If

    If wkbFonds Is Nothing Then
        Call VBA.Err.Raise( _
            ERROR_FONDSLISTE, METHOD_NAME, _
            "The Fondsliste workbook could not be accessed.")
    End If
    If wkbFonds.Worksheets.Count = 0 Then
        Call VBA.Err.Raise( _
            ERROR_FONDSLISTE, METHOD_NAME, _
            "The Fondsliste workbook does not contain a worksheet.")
    End If

    strCurrentOperation = "read Fondsliste worksheet"
    dtStepStarted = Now
    Set wksFonds = wkbFonds.Worksheets(1)
    lngLastRow = wksFonds.Cells(wksFonds.Rows.Count, 4).End(xlUp).Row

    If lngLastRow < FONDSLISTE_FIRST_DATA_ROW Then
        Call VBA.Err.Raise( _
            ERROR_FONDSLISTE, METHOD_NAME, _
            "No fund records were found in Fondsliste column D.")
    End If

    arrData = wksFonds.Range( _
        "A" & CStr(FONDSLISTE_FIRST_DATA_ROW) & ":O" & CStr(lngLastRow)).Value2
    Debug.Print "Dallas PERF | Fondsliste worksheet read | " & _
        CStr(DateDiff("s", dtStepStarted, Now)) & " s"

    strCurrentOperation = "build open-fund dictionary"
    dtStepStarted = Now
    For lngRow = 1 To UBound(arrData, 1)
        strStatus = Trim$(CStr(arrData(lngRow, 2)))
        strFundNumber = Trim$(CStr(arrData(lngRow, 4)))
        strTeam = Trim$(CStr(arrData(lngRow, 15)))

        If StrComp(strStatus, "Open", vbTextCompare) = 0 Then
            If Len(strFundNumber) > 0 Then dictTeams(strFundNumber) = strTeam
        End If
    Next lngRow
    Debug.Print "Dallas PERF | Fondsliste dictionary build (" & _
        CStr(dictTeams.Count) & " open funds) | " & _
        CStr(DateDiff("s", dtStepStarted, Now)) & " s"

ExitPoint:
    Set wksFonds = Nothing
    Set wkbAlreadyOpen = Nothing

    If blnCloseRequired Then
        blnCloseRequired = False
        dtStepStarted = Now
        cleanupErrNumber = CloseWorkbookSafely(wkbFonds, False)
        Debug.Print "Dallas PERF | Fondsliste Workbooks.Close | " & _
            CStr(DateDiff("s", dtStepStarted, Now)) & " s"

        If errNumber = 0 And cleanupErrNumber <> 0 Then
            errNumber = cleanupErrNumber
            errDescription = "The Fondsliste workbook could not be closed during cleanup."
        End If
    End If

    Set wkbFonds = Nothing

    If Len(strStagedCopyPath) > 0 Then
        dtStepStarted = Now
        cleanupErrNumber = DeleteStagedFondslisteFileSafely(strStagedCopyPath)
        Debug.Print "Dallas PERF | Fondsliste delete local copy | " & _
            CStr(DateDiff("s", dtStepStarted, Now)) & " s"
        If errNumber = 0 And cleanupErrNumber <> 0 Then
            errNumber = cleanupErrNumber
            errDescription = "The temporary Fondsliste file could not be deleted."
        End If
    End If

    If errNumber = 0 Then Set BuildOpenFundTeamDictionary = dictTeams
    Set dictTeams = Nothing
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    handlerErrNumber = VBA.Err.Number
    handlerErrDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, handlerErrNumber, handlerErrDescription, _
        "operation;row;localCopy", _
        strCurrentOperation, lngRow, blnUseLocalCopy)

    If errNumber = 0 Then
        errNumber = handlerErrNumber
        errDescription = "Operation: " & strCurrentOperation & ". " & handlerErrDescription
    End If

    GoTo ExitPoint
End Function


'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-29
' Parameters:    strSourcePath; strTargetPath; lngCopyErrNumber (output)
' Returns:       Boolean
' Description:   Attempts optional local staging with FileSystemObject.CopyFile.
'                A failed staging attempt is logged and reported to the caller;
'                the caller can safely open the same source workbook read-only.
'-------------------------------------------------------------------------------
Private Function TryStageFondslisteCopy( _
    ByVal strSourcePath As String, _
    ByVal strTargetPath As String, _
    ByRef lngCopyErrNumber As Long) As Boolean

    Const METHOD_NAME As String = "TryStageFondslisteCopy"
    Dim errDescription As String
    Dim errNumber As Long
    Dim objFileSystem As Object

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set objFileSystem = CreateObject("Scripting.FileSystemObject")
    Call objFileSystem.CopyFile(strSourcePath, strTargetPath, False)

    If Not objFileSystem.FileExists(strTargetPath) Then
        Call VBA.Err.Raise( _
            ERROR_FONDSLISTE, METHOD_NAME, "Local Fondsliste copy was not created.")
    End If
    If objFileSystem.GetFile(strTargetPath).Size = 0 Then
        Call VBA.Err.Raise( _
            ERROR_FONDSLISTE, METHOD_NAME, "Local Fondsliste copy is empty.")
    End If

    TryStageFondslisteCopy = True

ExitPoint:
    Set objFileSystem = Nothing
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    lngCopyErrNumber = errNumber
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "operation", "optional local Fondsliste copy")
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-29
' Parameters:    strFilePath As String
' Returns:       Long (0 = successful cleanup)
' Description:   Removes a staged Fondsliste without masking a previous error.
'-------------------------------------------------------------------------------
Private Function DeleteStagedFondslisteFileSafely(ByVal strFilePath As String) As Long
    Const METHOD_NAME As String = "DeleteStagedFondslisteFileSafely"
    Dim errDescription As String
    Dim errNumber As Long
    Dim objFileSystem As Object

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set objFileSystem = CreateObject("Scripting.FileSystemObject")
    If objFileSystem.FileExists(strFilePath) Then
        Call objFileSystem.DeleteFile(strFilePath, True)
    End If

ExitPoint:
    Set objFileSystem = Nothing
    DeleteStagedFondslisteFileSafely = errNumber
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "filePath", strFilePath)
    GoTo ExitPoint
End Function
