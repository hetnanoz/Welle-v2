Option Explicit

Private Const CLASS_NAME As String = "modMain"
Private Const SHOW_STAGE_TIMINGS As Boolean = True

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-27
' Parameters:    ---
' Returns:       ---
' Description:   Runs the complete Dallas Cash Transactions process.
'-------------------------------------------------------------------------------
Public Sub RunDallasCashTransactions()
    Const METHOD_NAME As String = "RunDallasCashTransactions"
    Dim appConfig As TAppConfig
    Dim arrCombined As Variant
    Dim blnEmailDraftCreated As Boolean
    Dim blnEmailLabelApplied As Boolean
    Dim cleanupErrNumber As Long
    Dim enableEventsCleanupErrNumber As Long
    Dim blnPreviousAskToUpdateLinks As Boolean
    Dim blnPreviousDisplayAlerts As Boolean
    Dim blnPreviousEnableEvents As Boolean
    Dim blnPreviousScreenUpdating As Boolean
    Dim blnStateCaptured As Boolean
    Dim colAttachmentPaths As Collection
    Dim colDestinationRoots As Collection
    Dim colTeams As Collection
    Dim dictAccounts As Object
    Dim dictAttachmentSubjects As Object
    Dim dictFundTeams As Object
    Dim dictProcessedMailSubjects As Object
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngArchivedMailCount As Long
    Dim lngPreviousAutomationSecurity As Long
    Dim strCurrentStage As String
    Dim strInputFolder As String
    Dim strLatestFondslistePath As String
    Dim strResolvedLocalOutputFolder As String
    Dim strResolvedSharePointOutputFolder As String
    Dim strStageWorkspace As String
    Dim strSuccessMessage As String
    Dim varPreviousStatusBar As Variant
    Dim xlPreviousCalculation As XlCalculation
    Dim dtRunTimestamp As Date
    Dim dtStageStarted As Date
    Dim strStageTimings As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    strCurrentStage = "capturing Excel application state"
    blnPreviousScreenUpdating = Application.ScreenUpdating
    blnPreviousEnableEvents = Application.EnableEvents
    blnPreviousDisplayAlerts = Application.DisplayAlerts
    blnPreviousAskToUpdateLinks = Application.AskToUpdateLinks
    xlPreviousCalculation = Application.Calculation
    varPreviousStatusBar = Application.StatusBar

    If VarType(varPreviousStatusBar) = vbString Then
        If InStr(1, CStr(varPreviousStatusBar), "Dallas Cash Transactions:", vbTextCompare) = 1 Then
            varPreviousStatusBar = False
        End If
    End If

    lngPreviousAutomationSecurity = Application.AutomationSecurity
    blnStateCaptured = True

    strCurrentStage = "applying Excel performance settings"
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.DisplayAlerts = False
    Application.AskToUpdateLinks = False
    Application.Calculation = xlCalculationManual
    Application.AutomationSecurity = AUTOMATION_SECURITY_FORCE_DISABLE
    Application.StatusBar = "Dallas Cash Transactions: reading configuration..."

    Call BeginTimedStage( _
        strCurrentStage, strStageTimings, dtStageStarted, _
        "reading configuration")
    dtRunTimestamp = Now
    Call LoadConfiguration(appConfig)
    Call BeginTimedStage( _
        strCurrentStage, strStageTimings, dtStageStarted, _
        "creating run workspace")
    strStageWorkspace = CreateRunWorkspace(dtRunTimestamp)
    strInputFolder = CreateInputWorkingFolder(appConfig, dtRunTimestamp, strStageWorkspace)

    Call BeginTimedStage( _
        strCurrentStage, strStageTimings, dtStageStarted, _
        "downloading Outlook attachments")
    Application.StatusBar = _
        "Dallas Cash Transactions: downloading Outlook attachments to daily Input folder..."
    Set colAttachmentPaths = DownloadCashTransactionAttachments( _
        appConfig, strInputFolder, dictAttachmentSubjects, dictProcessedMailSubjects)

    Call BeginTimedStage( _
        strCurrentStage, strStageTimings, dtStageStarted, _
        "merging input files")
    Application.StatusBar = "Dallas Cash Transactions: merging input files from disk..."
    arrCombined = MergeInputFiles(colAttachmentPaths, appConfig.InputWorksheetName, dictAttachmentSubjects)

    Call BeginTimedStage( _
        strCurrentStage, strStageTimings, dtStageStarted, _
        "deduplicating current transactions")
    Application.StatusBar = "Dallas Cash Transactions: removing duplicate transactions..."
    arrCombined = DeduplicateCurrentTransactions(arrCombined)

    Call BeginTimedStage( _
        strCurrentStage, strStageTimings, dtStageStarted, _
        "loading latest Fondsliste")
    Application.StatusBar = "Dallas Cash Transactions: loading the latest Fondsliste..."
    strLatestFondslistePath = GetLatestFondslistePath(appConfig)
    Set dictFundTeams = BuildOpenFundTeamDictionary( _
        strLatestFondslistePath, appConfig.FondslisteLocalCopy, strStageWorkspace)

    Call BeginTimedStage( _
        strCurrentStage, strStageTimings, dtStageStarted, _
        "updating Static_Data and enriching transactions")
    Application.StatusBar = "Dallas Cash Transactions: updating Static_Data..."
    Call UpdateStaticDataFromFondsliste(dictFundTeams)
    Set dictAccounts = BuildAccountMappingDictionary()
    Call EnrichTransactions(arrCombined, dictAccounts)

    Call BeginTimedStage( _
        strCurrentStage, strStageTimings, dtStageStarted, _
        "building output destinations")
    Set colTeams = ParseSupportedTeams(appConfig.SupportedTeams)
    Set colDestinationRoots = BuildDestinationRoots(appConfig)

    If appConfig.SaveLocal Then
        strResolvedLocalOutputFolder = BuildDatedFolderPath( _
            ResolveOutputPath(appConfig.OutputLocalBase), DateValue(dtRunTimestamp))
    End If

    If appConfig.SaveSharePoint Then
        strResolvedSharePointOutputFolder = BuildDatedFolderPath( _
            ResolveOutputPath(appConfig.OutputSharePointBase), DateValue(dtRunTimestamp))
    End If

    If appConfig.ArchiveInputFiles Then
        Call BeginTimedStage( _
            strCurrentStage, strStageTimings, dtStageStarted, _
            "archiving source input files")
        Application.StatusBar = _
            "Dallas Cash Transactions: archiving source input files..."
        Call PublishInputFilesToDailyFolders( _
            colAttachmentPaths, colDestinationRoots, _
            DateValue(dtRunTimestamp), strInputFolder)
    End If

    Call BeginTimedStage( _
        strCurrentStage, strStageTimings, dtStageStarted, _
        "creating daily reports")
    Application.StatusBar = "Dallas Cash Transactions: creating daily reports..."
    Call CreateDailyReports( _
        arrCombined, colTeams, colDestinationRoots, dtRunTimestamp, _
        strStageWorkspace, appConfig.CreateTeamFiles)

    If appConfig.UpdateHistorical Then
        Call BeginTimedStage( _
            strCurrentStage, strStageTimings, dtStageStarted, _
            "updating historical master files")
        Application.StatusBar = "Dallas Cash Transactions: updating historical master files..."
        Call UpdateHistoricalFiles( _
            arrCombined, colTeams, colDestinationRoots, _
            appConfig.CreateTeamFiles)
    End If

    If appConfig.CreateEmailDraft Then
        Call BeginTimedStage( _
            strCurrentStage, strStageTimings, dtStageStarted, _
            "creating Outlook email draft")
        Application.StatusBar = "Dallas Cash Transactions: creating Outlook email draft..."
        Call CreateCashTransactionsEmailDraft( _
            appConfig, arrCombined, colTeams, dtRunTimestamp, blnEmailLabelApplied)
        blnEmailDraftCreated = True
    End If

    If appConfig.ArchiveOutlookMessages Then
        Call BeginTimedStage( _
            strCurrentStage, strStageTimings, dtStageStarted, _
            "archiving processed Outlook messages")
        Application.StatusBar = _
            "Dallas Cash Transactions: archiving processed Outlook messages..."
        lngArchivedMailCount = ArchiveProcessedOutlookMessages( _
            appConfig, dictProcessedMailSubjects)
    End If

    Call BeginTimedStage( _
        strCurrentStage, strStageTimings, dtStageStarted, _
        "cleaning temporary workspace")
    If Len(strStageWorkspace) > 0 Then
        Call DeleteFolderIfExists(strStageWorkspace)
        strStageWorkspace = vbNullString
    End If

    Call BeginTimedStage( _
        strCurrentStage, strStageTimings, dtStageStarted, _
        "building success summary")
    strSuccessMessage = BuildSuccessMessage( _
        arrCombined, colTeams, colAttachmentPaths.Count, _
        lngArchivedMailCount, appConfig.ArchiveInputFiles, _
        appConfig.ArchiveOutlookMessages, appConfig.CreateEmailDraft, _
        blnEmailDraftCreated, blnEmailLabelApplied, _
        strResolvedLocalOutputFolder, strResolvedSharePointOutputFolder)

    Call CompleteTimedStage(strCurrentStage, strStageTimings, dtStageStarted)
    If SHOW_STAGE_TIMINGS Then
        strSuccessMessage = strSuccessMessage & vbCrLf & vbCrLf & _
            "Stage timings (seconds):" & strStageTimings
    End If

ExitPoint:
    Set dictAccounts = Nothing
    Set dictAttachmentSubjects = Nothing
    Set dictFundTeams = Nothing
    Set dictProcessedMailSubjects = Nothing
    Set colDestinationRoots = Nothing
    Set colTeams = Nothing
    Set colAttachmentPaths = Nothing

    If blnStateCaptured Then
        enableEventsCleanupErrNumber = RestoreEnableEventsSafely( _
            blnPreviousEnableEvents, errNumber = 0)

        cleanupErrNumber = RestoreApplicationStateSafely( _
            varPreviousStatusBar, lngPreviousAutomationSecurity, _
            xlPreviousCalculation, blnPreviousAskToUpdateLinks, _
            blnPreviousDisplayAlerts, blnPreviousScreenUpdating, _
            errNumber = 0)

        If errNumber = 0 And enableEventsCleanupErrNumber <> 0 Then
            errNumber = enableEventsCleanupErrNumber
            errDescription = "Excel EnableEvents state could not be restored."
        ElseIf errNumber = 0 And cleanupErrNumber <> 0 Then
            errNumber = cleanupErrNumber
            errDescription = "Excel application state could not be fully restored."
        End If
    End If

    If errNumber <> 0 Then
        Call ErrorManager.display
    ElseIf Len(strSuccessMessage) > 0 Then
        Call MsgBox(strSuccessMessage, vbInformation, "Dallas Cash Transactions")
    End If

    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description

    If Len(Trim$(errDescription)) = 0 Then
        errDescription = "Unspecified VBA error."
    End If

    If Len(strCurrentStage) = 0 Then strCurrentStage = "initialization"
    errDescription = _
        "Stage: " & strCurrentStage & ". Error " & _
        CStr(errNumber) & ": " & errDescription

    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "stage;inputFolder;stageWorkspace;fondslistePath", _
        strCurrentStage, strInputFolder, strStageWorkspace, _
        strLatestFondslistePath)
    GoTo ExitPoint
End Sub


'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-29
' Parameters:    ---
' Returns:       ---
' Description:   Diagnoses Fondsliste speed without processing Outlook or reports.
'-------------------------------------------------------------------------------
Public Sub BenchmarkFondslisteOnly()
    Const METHOD_NAME As String = "BenchmarkFondslisteOnly"
    Dim appConfig As TAppConfig
    Dim blnPreviousAskToUpdateLinks As Boolean
    Dim blnPreviousDisplayAlerts As Boolean
    Dim blnPreviousEnableEvents As Boolean
    Dim blnPreviousScreenUpdating As Boolean
    Dim blnStateCaptured As Boolean
    Dim cleanupErrNumber As Long
    Dim dictFundTeams As Object
    Dim dtBenchmarkStarted As Date
    Dim enableEventsCleanupErrNumber As Long
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngOpenFundCount As Long
    Dim lngPreviousAutomationSecurity As Long
    Dim lngTotalSeconds As Long
    Dim strCurrentStage As String
    Dim strLatestFondslistePath As String
    Dim strStageWorkspace As String
    Dim varPreviousStatusBar As Variant
    Dim xlPreviousCalculation As XlCalculation

    If Not DEV_MODE Then On Error GoTo ErrHandler

    strCurrentStage = "capture Excel state"
    blnPreviousScreenUpdating = Application.ScreenUpdating
    blnPreviousEnableEvents = Application.EnableEvents
    blnPreviousDisplayAlerts = Application.DisplayAlerts
    blnPreviousAskToUpdateLinks = Application.AskToUpdateLinks
    xlPreviousCalculation = Application.Calculation
    varPreviousStatusBar = Application.StatusBar
    lngPreviousAutomationSecurity = Application.AutomationSecurity
    blnStateCaptured = True

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.DisplayAlerts = False
    Application.AskToUpdateLinks = False
    Application.Calculation = xlCalculationManual
    Application.AutomationSecurity = AUTOMATION_SECURITY_FORCE_DISABLE
    Application.StatusBar = "Dallas: testing Fondsliste only..."

    dtBenchmarkStarted = Now
    strCurrentStage = "read Mapping"
    Call LoadConfiguration(appConfig)
    strCurrentStage = "create temporary workspace"
    strStageWorkspace = CreateRunWorkspace(Now)
    strCurrentStage = "find latest Fondsliste"
    strLatestFondslistePath = GetLatestFondslistePath(appConfig)
    strCurrentStage = "open and read Fondsliste"
    Set dictFundTeams = BuildOpenFundTeamDictionary( _
        strLatestFondslistePath, appConfig.FondslisteLocalCopy, strStageWorkspace)

    lngOpenFundCount = dictFundTeams.Count
    lngTotalSeconds = DateDiff("s", dtBenchmarkStarted, Now)
    Debug.Print "Dallas PERF | Fondsliste-only overall test | " & _
        CStr(lngTotalSeconds) & " s"

ExitPoint:
    Set dictFundTeams = Nothing

    If Len(strStageWorkspace) > 0 Then
        cleanupErrNumber = DeleteDiagnosticWorkspaceSafely(strStageWorkspace)
        If errNumber = 0 And cleanupErrNumber <> 0 Then
            errNumber = cleanupErrNumber
            errDescription = "Temporary Fondsliste test workspace could not be deleted."
        End If
    End If

    If blnStateCaptured Then
        enableEventsCleanupErrNumber = RestoreEnableEventsSafely( _
            blnPreviousEnableEvents, errNumber = 0)
        cleanupErrNumber = RestoreApplicationStateSafely( _
            varPreviousStatusBar, lngPreviousAutomationSecurity, _
            xlPreviousCalculation, blnPreviousAskToUpdateLinks, _
            blnPreviousDisplayAlerts, blnPreviousScreenUpdating, errNumber = 0)

        If errNumber = 0 And enableEventsCleanupErrNumber <> 0 Then
            errNumber = enableEventsCleanupErrNumber
            errDescription = "Could not restore Excel EnableEvents."
        ElseIf errNumber = 0 And cleanupErrNumber <> 0 Then
            errNumber = cleanupErrNumber
            errDescription = "Could not restore Excel application state."
        End If
    End If

    If errNumber <> 0 Then
        Call ErrorManager.display
    Else
        Call MsgBox( _
            "Fondsliste-only diagnostic completed." & vbCrLf & _
            "Total time: " & CStr(lngTotalSeconds) & " s" & vbCrLf & _
            "Open funds: " & CStr(lngOpenFundCount) & vbCrLf & _
            "Local copy enabled: " & CStr(appConfig.FondslisteLocalCopy) & vbCrLf & _
            "See Immediate window (Ctrl+G) for individual steps.", _
            vbInformation, "Dallas Fondsliste test")
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "stage", strCurrentStage)
    GoTo ExitPoint
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-29
' Parameters:    strWorkspace As String
' Returns:       Long (0 = successful cleanup)
' Description:   Cleans temporary diagnostics folder without masking errors.
'-------------------------------------------------------------------------------
Private Function DeleteDiagnosticWorkspaceSafely(ByVal strWorkspace As String) As Long
    Const METHOD_NAME As String = "DeleteDiagnosticWorkspaceSafely"
    Dim errDescription As String
    Dim errNumber As Long
    Dim objFileSystem As Object

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set objFileSystem = CreateObject("Scripting.FileSystemObject")
    If objFileSystem.FolderExists(strWorkspace) Then
        Call objFileSystem.DeleteFolder(strWorkspace, True)
    End If

ExitPoint:
    Set objFileSystem = Nothing
    DeleteDiagnosticWorkspaceSafely = errNumber
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "workspace", strWorkspace)
    GoTo ExitPoint
End Function


'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-29
' Parameters:    Current stage, log, stage start and next stage.
' Returns:       ---
' Description:   Records completed stage duration without creating disk log files.
'-------------------------------------------------------------------------------
Private Sub BeginTimedStage( _
    ByRef strCurrentStage As String, _
    ByRef strStageTimings As String, _
    ByRef dtStageStarted As Date, _
    ByVal strNewStage As String)

    Const METHOD_NAME As String = "BeginTimedStage"
    Dim errDescription As String
    Dim errNumber As Long

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Call CompleteTimedStage(strCurrentStage, strStageTimings, dtStageStarted)
    strCurrentStage = strNewStage
    dtStageStarted = Now

ExitPoint:
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "stage", strNewStage)
    GoTo ExitPoint
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-29
' Parameters:    Current stage, log and stage start.
' Returns:       ---
' Description:   Appends elapsed seconds; does not log business or personal data.
'-------------------------------------------------------------------------------
Private Sub CompleteTimedStage( _
    ByVal strCurrentStage As String, _
    ByRef strStageTimings As String, _
    ByVal dtStageStarted As Date)

    Const METHOD_NAME As String = "CompleteTimedStage"
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngElapsedSeconds As Long
    Dim strShortStage As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    If dtStageStarted <> 0 And SHOW_STAGE_TIMINGS Then
        lngElapsedSeconds = DateDiff("s", dtStageStarted, Now)
        Debug.Print "Dallas PERF | " & strCurrentStage & " | " & _
            CStr(lngElapsedSeconds) & " s"

        Select Case strCurrentStage
            Case "downloading Outlook attachments"
                strShortStage = "Outlook"
            Case "merging input files"
                strShortStage = "Merge"
            Case "loading latest Fondsliste"
                strShortStage = "Fondsliste"
            Case "updating Static_Data and enriching transactions"
                strShortStage = "Static_Data"
            Case "building output destinations"
                strShortStage = "Output paths"
            Case "archiving source input files"
                strShortStage = "Input archive"
            Case "creating daily reports"
                strShortStage = "Daily reports"
            Case "updating historical master files"
                strShortStage = "Historical"
            Case "creating Outlook email draft"
                strShortStage = "Mail draft"
            Case "archiving processed Outlook messages"
                strShortStage = "Mail archive"
        End Select

        If Len(strShortStage) > 0 Then
            strStageTimings = strStageTimings & vbCrLf & _
                strShortStage & ": " & CStr(lngElapsedSeconds)
        End If
    End If

ExitPoint:
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "stage", strCurrentStage)
    GoTo ExitPoint
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-27
' Parameters:    Combined data, Teams, counts, email status and output folders.
' Returns:       String
' Description:   Builds the final success message with transaction counts per team.
'-------------------------------------------------------------------------------
Private Function BuildSuccessMessage( _
    ByRef arrCombined As Variant, _
    ByVal colTeams As Collection, _
    ByVal lngAttachmentCount As Long, _
    ByVal lngArchivedMailCount As Long, _
    ByVal blnArchiveInputFiles As Boolean, _
    ByVal blnArchiveOutlookMessages As Boolean, _
    ByVal blnEmailRequested As Boolean, _
    ByVal blnEmailDraftCreated As Boolean, _
    ByVal blnEmailLabelApplied As Boolean, _
    ByVal strLocalOutputFolder As String, _
    ByVal strSharePointOutputFolder As String) As String
    Const METHOD_NAME As String = "BuildSuccessMessage"
    Dim arrKeys As Variant
    Dim dictConfigured As Object
    Dim dictTeamCounts As Object
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngCount As Long
    Dim lngKey As Long
    Dim lngOtherCount As Long
    Dim lngRow As Long
    Dim lngTeam As Long
    Dim lngTransactionCount As Long
    Dim lngUnknownCount As Long
    Dim strOtherTeams As String
    Dim strResult As String
    Dim strTeam As String
    Dim strTeamsWithTrades As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set dictConfigured = CreateObject("Scripting.Dictionary")
    dictConfigured.CompareMode = vbTextCompare
    Set dictTeamCounts = CreateObject("Scripting.Dictionary")
    dictTeamCounts.CompareMode = vbTextCompare

    lngTransactionCount = UBound(arrCombined, 1) - 1
    lngUnknownCount = CountUnknownTransactions(arrCombined)

    For lngRow = 2 To UBound(arrCombined, 1)
        strTeam = Trim$(CStr(arrCombined(lngRow, 5)))
        If Len(strTeam) = 0 Then strTeam = "UNKNOWN TEAM"

        If dictTeamCounts.Exists(strTeam) Then
            dictTeamCounts(strTeam) = CLng(dictTeamCounts(strTeam)) + 1
        Else
            dictTeamCounts.Add strTeam, 1
        End If
    Next lngRow

    For lngTeam = 1 To colTeams.Count
        strTeam = Trim$(CStr(colTeams(lngTeam)))
        If Len(strTeam) > 0 Then dictConfigured(strTeam) = True

        If dictTeamCounts.Exists(strTeam) Then
            lngCount = CLng(dictTeamCounts(strTeam))

            If Len(strTeamsWithTrades) > 0 Then strTeamsWithTrades = strTeamsWithTrades & ", "
            strTeamsWithTrades = strTeamsWithTrades & strTeam & " (" & CStr(lngCount) & ")"
        End If
    Next lngTeam

    If dictTeamCounts.Count > 0 Then
        arrKeys = dictTeamCounts.Keys

        For lngKey = LBound(arrKeys) To UBound(arrKeys)
            strTeam = CStr(arrKeys(lngKey))

            If Not dictConfigured.Exists(strTeam) Then
                lngCount = CLng(dictTeamCounts(strTeam))
                lngOtherCount = lngOtherCount + lngCount

                If Len(strOtherTeams) > 0 Then strOtherTeams = strOtherTeams & ", "
                strOtherTeams = strOtherTeams & strTeam & " (" & CStr(lngCount) & ")"
            End If
        Next lngKey
    End If

    If Len(strTeamsWithTrades) = 0 Then strTeamsWithTrades = "none"

    strResult = "Dallas Cash Transactions completed successfully."
    strResult = strResult & vbCrLf & vbCrLf
    strResult = strResult & "Unique transactions: " & CStr(lngTransactionCount) & vbCrLf
    strResult = strResult & _
        "Excel attachments processed: " & CStr(lngAttachmentCount) & vbCrLf

    If blnArchiveInputFiles Then
        strResult = strResult & "Input file archive: enabled" & vbCrLf
    Else
        strResult = strResult & "Input file archive: skipped" & vbCrLf
    End If

    If blnArchiveOutlookMessages Then
        strResult = strResult & _
            "Outlook messages archived: " & CStr(lngArchivedMailCount) & vbCrLf
    Else
        strResult = strResult & "Outlook message archive: skipped" & vbCrLf
    End If

    strResult = strResult & "Teams with trades: " & strTeamsWithTrades & vbCrLf
    strResult = strResult & "Unknown transactions: " & CStr(lngUnknownCount)

    If blnEmailRequested Then
        strResult = strResult & vbCrLf

        If blnEmailDraftCreated Then
            strResult = strResult & "Outlook email draft: created"
        Else
            strResult = strResult & "Outlook email draft: not created"
        End If

        strResult = strResult & vbCrLf

        If blnEmailLabelApplied Then
            strResult = strResult & "Outlook Internal label: applied"
        Else
            strResult = strResult & "Outlook Internal label: not applied (best effort)"
        End If
    End If

    If lngOtherCount > 0 Then
        strResult = strResult & vbCrLf
        strResult = strResult & "Other / unconfigured Team values: " & strOtherTeams
    End If

    If Len(strLocalOutputFolder) > 0 Then
        strResult = strResult & vbCrLf & vbCrLf
        strResult = strResult & "Local output:" & vbCrLf & strLocalOutputFolder
    End If

    If Len(strSharePointOutputFolder) > 0 Then
        strResult = strResult & vbCrLf & vbCrLf
        strResult = strResult & "SharePoint output:" & vbCrLf & strSharePointOutputFolder
    End If

ExitPoint:
    Set dictTeamCounts = Nothing
    Set dictConfigured = Nothing

    If errNumber = 0 Then BuildSuccessMessage = strResult
    If errNumber <> 0 Then Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "row", lngRow)
    GoTo ExitPoint
End Function


'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-28
' Parameters:    Previous EnableEvents state and whether cleanup errors may log.
' Returns:       Long
' Description:   Restores Excel events independently so another cleanup failure
'                cannot leave Application.EnableEvents disabled.
'-------------------------------------------------------------------------------
Private Function RestoreEnableEventsSafely( _
    ByVal blnPreviousEnableEvents As Boolean, _
    ByVal blnLogCleanupError As Boolean) As Long

    Const METHOD_NAME As String = "RestoreEnableEventsSafely"
    Dim errDescription As String
    Dim errNumber As Long

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Application.EnableEvents = blnPreviousEnableEvents

ExitPoint:
    RestoreEnableEventsSafely = errNumber
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    errDescription = _
        "Cleanup stage: restore Application.EnableEvents. Error " & _
        CStr(errNumber) & ": " & errDescription

    If blnLogCleanupError Then
        Call ErrorManager.addError( _
            CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    End If

    ' Cleanup helper: return the cleanup error rather than re-raising it.
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-20
' Parameters:    Previous Excel state plus whether cleanup errors may be logged.
' Returns:       Long
' Description:   Restores Excel state during MASTER cleanup without re-raising
'                a cleanup error that could mask a previously captured error.
'-------------------------------------------------------------------------------
Private Function RestoreApplicationStateSafely( _
    ByVal varPreviousStatusBar As Variant, _
    ByVal lngPreviousAutomationSecurity As Long, _
    ByVal xlPreviousCalculation As XlCalculation, _
    ByVal blnPreviousAskToUpdateLinks As Boolean, _
    ByVal blnPreviousDisplayAlerts As Boolean, _
    ByVal blnPreviousScreenUpdating As Boolean, _
    ByVal blnLogCleanupError As Boolean) As Long

    Const METHOD_NAME As String = "RestoreApplicationStateSafely"
    Dim errDescription As String
    Dim errNumber As Long
    Dim strCleanupStage As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    strCleanupStage = "restore Application.StatusBar"
    Application.StatusBar = varPreviousStatusBar
    strCleanupStage = "restore Application.AutomationSecurity"
    Application.AutomationSecurity = lngPreviousAutomationSecurity
    strCleanupStage = "restore Application.Calculation"
    Application.Calculation = xlPreviousCalculation
    strCleanupStage = "restore Application.AskToUpdateLinks"
    Application.AskToUpdateLinks = blnPreviousAskToUpdateLinks
    strCleanupStage = "restore Application.DisplayAlerts"
    Application.DisplayAlerts = blnPreviousDisplayAlerts
    strCleanupStage = "restore Application.ScreenUpdating"
    Application.ScreenUpdating = blnPreviousScreenUpdating

ExitPoint:
    RestoreApplicationStateSafely = errNumber
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    errDescription = _
        "Cleanup stage: " & strCleanupStage & ". Error " & _
        CStr(errNumber) & ": " & errDescription
    If blnLogCleanupError Then
        Call ErrorManager.addError( _
            CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
            "cleanupStage", strCleanupStage)
    End If

    ' Cleanup helper: return the cleanup error rather than re-raising it.
    ' When a primary MASTER error already exists, do not add a newer cleanup
    ' error to ErrorManager because that could hide the primary user-facing error.
    GoTo ExitPoint
End Function
