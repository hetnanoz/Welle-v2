Option Explicit

Private Const CLASS_NAME As String = "modStaticData"

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-26
' Parameters:    dictFundTeams As Object
' Returns:       ---
' Description:   Updates Static_Data Team values from Open Fondsliste funds.
'-------------------------------------------------------------------------------
Public Sub UpdateStaticDataFromFondsliste(ByVal dictFundTeams As Object)
    Const METHOD_NAME As String = "UpdateStaticDataFromFondsliste"
    Dim arrFundCodes As Variant
    Dim arrTeams As Variant
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngDataRows As Long
    Dim lngLastRow As Long
    Dim lngRow As Long
    Dim strFundNumber As String
    Dim wksStatic As Excel.Worksheet

    If Not DEV_MODE Then On Error GoTo ErrHandler

    If dictFundTeams Is Nothing Then
        Call VBA.Err.Raise( _
            ERROR_STATIC_DATA, METHOD_NAME, _
            "Fondsliste Team dictionary is not available.")
    End If

    Set wksStatic = ThisWorkbook.Worksheets(SHEET_STATIC_DATA)
    lngLastRow = GetStaticDataLastRow(wksStatic)

    If lngLastRow < STATIC_DATA_FIRST_DATA_ROW Then
        Call VBA.Err.Raise( _
            ERROR_STATIC_DATA, METHOD_NAME, _
            "Static_Data does not contain any data rows.")
    End If

    lngDataRows = lngLastRow - STATIC_DATA_FIRST_DATA_ROW + 1
    arrFundCodes = wksStatic.Range( _
        wksStatic.Cells(STATIC_DATA_FIRST_DATA_ROW, 3), _
        wksStatic.Cells(lngLastRow, 3)).Value2
    ReDim arrTeams(1 To lngDataRows, 1 To 1)

    For lngRow = 1 To lngDataRows
        strFundNumber = NormalizeLookupKey(arrFundCodes(lngRow, 1))

        If Len(strFundNumber) > 0 And dictFundTeams.Exists(strFundNumber) Then
            arrTeams(lngRow, 1) = CStr(dictFundTeams(strFundNumber))
        Else
            arrTeams(lngRow, 1) = vbNullString
        End If
    Next lngRow

    wksStatic.Range( _
        wksStatic.Cells(STATIC_DATA_FIRST_DATA_ROW, 7), _
        wksStatic.Cells(lngLastRow, 7)).Value2 = arrTeams

ExitPoint:
    Set wksStatic = Nothing
    If errNumber <> 0 Then Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError(CLASS_NAME, METHOD_NAME, errNumber, errDescription, "row", lngRow)
    GoTo ExitPoint
End Sub


'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-30
' Parameters:    ---
' Returns:       ---
' Description:   Reports duplicate Account mappings, including resolvable rules.
'-------------------------------------------------------------------------------
Public Sub DiagnoseStaticDataAccountConflicts()
    Const METHOD_NAME As String = "DiagnoseStaticDataAccountConflicts"
    Dim arrData As Variant
    Dim arrPreviousMapping As Variant
    Dim dictAccounts As Object
    Dim dictFirstRows As Object
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngConflictCount As Long
    Dim lngLastRow As Long
    Dim lngRow As Long
    Dim strConflictFields As String
    Dim strDisplayedIdentifier As String
    Dim strFund As String
    Dim strIdentifier As String
    Dim strIdentifierContext As String
    Dim strMaster As String
    Dim strTeam As String
    Dim wksStatic As Excel.Worksheet

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set wksStatic = ThisWorkbook.Worksheets(SHEET_STATIC_DATA)
    lngLastRow = GetStaticDataLastRow(wksStatic)
    If lngLastRow < STATIC_DATA_FIRST_DATA_ROW Then
        Call VBA.Err.Raise( _
            ERROR_STATIC_DATA, METHOD_NAME, "Static_Data has no account mappings.")
    End If

    arrData = wksStatic.Range( _
        wksStatic.Cells(STATIC_DATA_FIRST_DATA_ROW, 2), _
        wksStatic.Cells(lngLastRow, 7)).Value2
    Set dictAccounts = CreateObject("Scripting.Dictionary")
    dictAccounts.CompareMode = vbTextCompare
    Set dictFirstRows = CreateObject("Scripting.Dictionary")
    dictFirstRows.CompareMode = vbTextCompare

    Debug.Print "Dallas AUDIT | Static_Data duplicate Account check | starting"

    For lngRow = 1 To UBound(arrData, 1)
        If IsNumeric(arrData(lngRow, 1)) Then
            strDisplayedIdentifier = CStr( _
                wksStatic.Cells(STATIC_DATA_FIRST_DATA_ROW + lngRow - 1, 2).Text)
        ElseIf IsError(arrData(lngRow, 1)) Then
            strDisplayedIdentifier = vbNullString
        ElseIf IsNull(arrData(lngRow, 1)) Or IsEmpty(arrData(lngRow, 1)) Then
            strDisplayedIdentifier = vbNullString
        Else
            strDisplayedIdentifier = Trim$(CStr(arrData(lngRow, 1)))
        End If

        strIdentifierContext = "Static_Data row " & _
            CStr(STATIC_DATA_FIRST_DATA_ROW + lngRow - 1) & "."
        strIdentifier = NormalizeAccountIdentifier( _
            arrData(lngRow, 1), strDisplayedIdentifier, _
            ERROR_STATIC_DATA, strIdentifierContext)

        If Len(strIdentifier) > 0 Then
            strFund = GetTextValue(arrData(lngRow, 2))
            strMaster = GetOptionalMasterValue(arrData(lngRow, 3))
            strTeam = GetTextValue(arrData(lngRow, 6))

            If dictAccounts.Exists(strIdentifier) Then
                arrPreviousMapping = dictAccounts(strIdentifier)
                strConflictFields = vbNullString
                If StrComp(CStr(arrPreviousMapping(0)), strFund, vbTextCompare) <> 0 Then
                    strConflictFields = "C (Fund)"
                End If
                If StrComp(CStr(arrPreviousMapping(1)), strMaster, vbTextCompare) <> 0 Then
                    If Len(strConflictFields) > 0 Then strConflictFields = strConflictFields & ", "
                    strConflictFields = strConflictFields & "D (Master)"
                End If
                If StrComp(CStr(arrPreviousMapping(2)), strTeam, vbTextCompare) <> 0 Then
                    If Len(strConflictFields) > 0 Then strConflictFields = strConflictFields & ", "
                    strConflictFields = strConflictFields & "G (Team)"
                End If
                If Len(strConflictFields) > 0 Then
                    lngConflictCount = lngConflictCount + 1
                    Debug.Print "Dallas AUDIT | conflict rows " & _
                        CStr(dictFirstRows(strIdentifier)) & " and " & _
                        CStr(STATIC_DATA_FIRST_DATA_ROW + lngRow - 1) & _
                        " | different columns: " & strConflictFields
                End If
            Else
                dictAccounts.Add strIdentifier, Array(strFund, strMaster, strTeam)
                dictFirstRows.Add strIdentifier, STATIC_DATA_FIRST_DATA_ROW + lngRow - 1
            End If
        End If
    Next lngRow

    Debug.Print "Dallas AUDIT | Static_Data conflicting duplicate rows | " & _
        CStr(lngConflictCount)

ExitPoint:
    Set wksStatic = Nothing
    Set dictFirstRows = Nothing
    Set dictAccounts = Nothing
    If errNumber <> 0 Then Call ErrorManager.display
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, _
        "row", lngRow)
    GoTo ExitPoint
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-30
' Parameters:    ---
' Returns:       Object (Account -> Collection of candidate mapping rows)
' Description:   Reads Account mappings with optional per-row extra match fields.
'-------------------------------------------------------------------------------
Public Function BuildAccountMappingDictionary() As Object
    Const METHOD_NAME As String = "BuildAccountMappingDictionary"
    Dim arrData As Variant
    Dim colCandidates As Collection
    Dim dictAccounts As Object
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngColName As Long
    Dim lngColValue As Long
    Dim lngExtraColumn As Long
    Dim lngLastColumn As Long
    Dim lngLastRow As Long
    Dim lngRow As Long
    Dim strDisplayedIdentifier As String
    Dim strExtraName As String
    Dim strExtraValue As String
    Dim strIdentifier As String
    Dim strIdentifierContext As String
    Dim strFund As String
    Dim strMaster As String
    Dim strTeam As String
    Dim wksStatic As Excel.Worksheet

    If Not DEV_MODE Then On Error GoTo ErrHandler

    Set wksStatic = ThisWorkbook.Worksheets(SHEET_STATIC_DATA)
    lngLastRow = GetStaticDataLastRow(wksStatic)
    If lngLastRow < STATIC_DATA_FIRST_DATA_ROW Then
        Call VBA.Err.Raise(ERROR_STATIC_DATA, METHOD_NAME, _
            "Static_Data does not contain any account mappings.")
    End If

    Call GetStaticDataExceptionColumns(wksStatic, lngColName, lngColValue)
    lngLastColumn = 7
    If lngColName > lngLastColumn Then lngLastColumn = lngColName
    If lngColValue > lngLastColumn Then lngLastColumn = lngColValue

    arrData = wksStatic.Range( _
        wksStatic.Cells(STATIC_DATA_FIRST_DATA_ROW, 2), _
        wksStatic.Cells(lngLastRow, lngLastColumn)).Value2

    Set dictAccounts = CreateObject("Scripting.Dictionary")
    dictAccounts.CompareMode = vbTextCompare

    For lngRow = 1 To UBound(arrData, 1)
        If IsNumeric(arrData(lngRow, 1)) Then
            strDisplayedIdentifier = CStr(wksStatic.Cells( _
                STATIC_DATA_FIRST_DATA_ROW + lngRow - 1, 2).Text)
        ElseIf IsError(arrData(lngRow, 1)) Then
            strDisplayedIdentifier = vbNullString
        ElseIf IsNull(arrData(lngRow, 1)) Or IsEmpty(arrData(lngRow, 1)) Then
            strDisplayedIdentifier = vbNullString
        Else
            strDisplayedIdentifier = Trim$(CStr(arrData(lngRow, 1)))
        End If

        strIdentifierContext = "Static_Data row " & _
            CStr(STATIC_DATA_FIRST_DATA_ROW + lngRow - 1) & "."
        strIdentifier = NormalizeAccountIdentifier( _
            arrData(lngRow, 1), strDisplayedIdentifier, _
            ERROR_STATIC_DATA, strIdentifierContext)

        If Len(strIdentifier) > 0 Then
            strFund = GetTextValue(arrData(lngRow, 2))
            strMaster = GetOptionalMasterValue(arrData(lngRow, 3))
            strTeam = GetTextValue(arrData(lngRow, 6))
            lngExtraColumn = 0
            strExtraName = vbNullString
            strExtraValue = vbNullString

            If lngColName > 0 Then
                strExtraName = GetTextValue(arrData(lngRow, lngColName - 1))
                strExtraValue = GetTextValue(arrData(lngRow, lngColValue - 1))
                If (Len(strExtraName) = 0) <> (Len(strExtraValue) = 0) Then
                    Call VBA.Err.Raise(ERROR_STATIC_DATA, METHOD_NAME, _
                        "Both Exception Extra Column and Exception Extra Value " & _
                        "must be filled at " & strIdentifierContext)
                End If
                If Len(strExtraName) > 0 Then
                    lngExtraColumn = ResolveExceptionTransactionColumn(strExtraName)
                    If lngExtraColumn = 0 Then
                        Call VBA.Err.Raise(ERROR_STATIC_DATA, METHOD_NAME, _
                            "Unknown or unsupported Exception Extra Column at " & _
                            strIdentifierContext & " Use an input header or F:U letter.")
                    End If
                End If
            End If

            If dictAccounts.Exists(strIdentifier) Then
                Set colCandidates = dictAccounts(strIdentifier)
            Else
                Set colCandidates = New Collection
                dictAccounts.Add strIdentifier, colCandidates
            End If

            ' Candidate: Fund, Master, Team, transaction column, match value, Static row.
            colCandidates.Add Array( _
                strFund, strMaster, strTeam, lngExtraColumn, strExtraValue, _
                STATIC_DATA_FIRST_DATA_ROW + lngRow - 1)
        End If
    Next lngRow

    Debug.Print "Dallas PERF | Static_Data Account groups | " & CStr(dictAccounts.Count)

ExitPoint:
    Set colCandidates = Nothing
    Set wksStatic = Nothing
    If errNumber = 0 Then Set BuildAccountMappingDictionary = dictAccounts
    Set dictAccounts = Nothing
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, "row", lngRow)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-30
' Parameters:    arrCombined As Variant; dictAccounts As Object
' Returns:       ---
' Description:   Applies exact exception matching and quarantines ambiguity.
'-------------------------------------------------------------------------------
Public Sub EnrichTransactions(ByRef arrCombined As Variant, ByVal dictAccounts As Object)
    Const METHOD_NAME As String = "EnrichTransactions"
    Dim arrMapping As Variant
    Dim colCandidates As Collection
    Dim dtAddedTimestamp As Date
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngExceptionMatched As Long
    Dim lngRow As Long
    Dim lngUnknown As Long
    Dim strAccount As String
    Dim strResultKind As String
    Dim strUserId As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    If dictAccounts Is Nothing Then
        Call VBA.Err.Raise(ERROR_STATIC_DATA, METHOD_NAME, _
            "Account mapping dictionary is not available.")
    End If

    dtAddedTimestamp = Now
    strUserId = GetCurrentUserId()

    For lngRow = 2 To UBound(arrCombined, 1)
        strAccount = NormalizeLookupKey(arrCombined(lngRow, 6))
        strResultKind = "UNKNOWN"
        arrMapping = Array("UNKNOWN FUND", "UNKNOWN FUND", "UNKNOWN TEAM")

        If Len(strAccount) > 0 And dictAccounts.Exists(strAccount) Then
            Set colCandidates = dictAccounts(strAccount)
            arrMapping = ResolveAccountCandidateMapping( _
                colCandidates, arrCombined, lngRow, strResultKind)
        End If

        If strResultKind = "EXCEPTION" Then lngExceptionMatched = lngExceptionMatched + 1
        If strResultKind = "UNKNOWN" Then lngUnknown = lngUnknown + 1

        arrCombined(lngRow, 1) = arrMapping(0)
        arrCombined(lngRow, 2) = arrMapping(1)
        arrCombined(lngRow, 5) = arrMapping(2)
        arrCombined(lngRow, 3) = arrCombined(lngRow, 8)
        arrCombined(lngRow, 4) = arrCombined(lngRow, 9)
        arrCombined(lngRow, OUTPUT_TIMESTAMP_COLUMN) = dtAddedTimestamp
        arrCombined(lngRow, OUTPUT_USER_ID_COLUMN) = strUserId
    Next lngRow

    Debug.Print "Dallas PERF | Static_Data exception matches | " & _
        CStr(lngExceptionMatched)
    Debug.Print "Dallas PERF | Static_Data unresolved transactions | " & _
        CStr(lngUnknown)

ExitPoint:
    Set colCandidates = Nothing
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, "row", lngRow)
    GoTo ExitPoint
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-30
' Parameters:    colCandidates, arrCombined, lngRow, strResultKind
' Returns:       Variant (Fund, Master, Team)
' Description:   Exception wins; conflicting unmatched Accounts become UNKNOWN.
'-------------------------------------------------------------------------------
Private Function ResolveAccountCandidateMapping( _
    ByVal colCandidates As Collection, _
    ByRef arrCombined As Variant, _
    ByVal lngRow As Long, _
    ByRef strResultKind As String) As Variant

    Const METHOD_NAME As String = "ResolveAccountCandidateMapping"
    Dim arrCandidate As Variant
    Dim arrFirst As Variant
    Dim arrResolved As Variant
    Dim blnConflict As Boolean
    Dim blnMatchFound As Boolean
    Dim blnMatchConflict As Boolean
    Dim blnDefaultFound As Boolean
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngMatchColumn As Long
    Dim strTransactionValue As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    arrResolved = Array("UNKNOWN FUND", "UNKNOWN FUND", "UNKNOWN TEAM")
    strResultKind = "UNKNOWN"
    arrFirst = colCandidates(1)

    ' Determine whether Account alone maps to competing Fund/Master/Team values.
    For Each arrCandidate In colCandidates
        If Not CandidateMappingsEqual(arrFirst, arrCandidate) Then
            blnConflict = True
        End If
    Next arrCandidate

    ' The more-specific matching exception is considered first.
    For Each arrCandidate In colCandidates
        lngMatchColumn = CLng(arrCandidate(3))
        If lngMatchColumn > 0 Then
            strTransactionValue = GetTransactionExceptionValue( _
                arrCombined(lngRow, lngMatchColumn))
            If StrComp(strTransactionValue, CStr(arrCandidate(4)), vbTextCompare) = 0 Then
                If blnMatchFound Then
                    If Not CandidateMappingsEqual(arrResolved, arrCandidate) Then
                        blnMatchConflict = True
                    End If
                Else
                    arrResolved = Array(arrCandidate(0), arrCandidate(1), arrCandidate(2))
                    blnMatchFound = True
                End If
            End If
        End If
    Next arrCandidate

    If blnMatchFound Then
        If blnMatchConflict Then
            arrResolved = Array("UNKNOWN FUND", "UNKNOWN FUND", "UNKNOWN TEAM")
        Else
            strResultKind = "EXCEPTION"
        End If
    ElseIf Not blnConflict Then
        ' Only an unconditional row can serve as a default mapping.
        For Each arrCandidate In colCandidates
            If CLng(arrCandidate(3)) = 0 Then
                arrResolved = Array(arrCandidate(0), arrCandidate(1), arrCandidate(2))
                blnDefaultFound = True
                Exit For
            End If
        Next arrCandidate
        If blnDefaultFound Then strResultKind = "DEFAULT"
    End If

    If strResultKind <> "UNKNOWN" Then
        If Len(Trim$(CStr(arrResolved(0)))) = 0 Then arrResolved(0) = "UNKNOWN FUND"
        If Len(Trim$(CStr(arrResolved(2)))) = 0 Then arrResolved(2) = "UNKNOWN TEAM"
    End If

ExitPoint:
    If errNumber = 0 Then ResolveAccountCandidateMapping = arrResolved
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError( _
        CLASS_NAME, METHOD_NAME, errNumber, errDescription, "row", lngRow)
    GoTo ExitPoint
End Function

' Author:        Pawel Ligezka

' Parameters:    arrA, arrB As Variant
' Returns:       Boolean
' Description:   Compares candidate mappings (first three elements only).

    Const METHOD_NAME As String = "CandidateMappingsEqual"
    Dim blnEqual As Boolean
    Dim errDescription As String
    Dim errNumber As Long

    If Not DEV_MODE Then On Error GoTo ErrHandler

    blnEqual = _
        StrComp(CStr(arrA(0)), CStr(arrB(0)), vbTextCompare) = 0 And _
        StrComp(CStr(arrA(1)), CStr(arrB(1)), vbTextCompare) = 0 And _
        StrComp(CStr(arrA(2)), CStr(arrB(2)), vbTextCompare) = 0

ExitPoint:
    If errNumber = 0 Then CandidateMappingsEqual = blnEqual
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-30
' Parameters:    wksStatic, lngColName, lngColValue
' Returns:       ---
' Description:   Finds optional exception headers beyond existing column G.
'-------------------------------------------------------------------------------
Private Sub GetStaticDataExceptionColumns( _
    ByVal wksStatic As Excel.Worksheet, _
    ByRef lngColName As Long, _
    ByRef lngColValue As Long)

    Const METHOD_NAME As String = "GetStaticDataExceptionColumns"
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngColumn As Long
    Dim lngLastHeaderColumn As Long
    Dim strHeader As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    lngLastHeaderColumn = wksStatic.Cells(1, wksStatic.Columns.Count).End(xlToLeft).Column
    For lngColumn = 8 To lngLastHeaderColumn
        strHeader = GetTextValue(wksStatic.Cells(1, lngColumn).Value2)
        Select Case UCase$(strHeader)
            Case "EXCEPTION EXTRA COLUMN"
                If lngColName <> 0 Then
                    Call VBA.Err.Raise(ERROR_STATIC_DATA, METHOD_NAME, _
                        "Duplicate Exception Extra Column header in Static_Data.")
                End If
                lngColName = lngColumn
            Case "EXCEPTION EXTRA VALUE"
                If lngColValue <> 0 Then
                    Call VBA.Err.Raise(ERROR_STATIC_DATA, METHOD_NAME, _
                        "Duplicate Exception Extra Value header in Static_Data.")
                End If
                lngColValue = lngColumn
        End Select
    Next lngColumn

    If (lngColName = 0) <> (lngColValue = 0) Then
        Call VBA.Err.Raise(ERROR_STATIC_DATA, METHOD_NAME, _
            "Both Exception Extra Column and Exception Extra Value " & _
            "headers are required in Static_Data.")
    End If

ExitPoint:
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Sub

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-10-30
' Parameters:    strColumnName As String
' Returns:       Long (F:U, input transaction columns)
' Description:   Resolves either an output header name or Excel column letter.
'-------------------------------------------------------------------------------
Private Function ResolveExceptionTransactionColumn(ByVal strColumnName As String) As Long
    Const METHOD_NAME As String = "ResolveExceptionTransactionColumn"
    Dim arrHeaders As Variant
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngColumn As Long
    Dim lngFound As Long
    Dim strKey As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    strKey = UCase$(Trim$(strColumnName))
    arrHeaders = GetOutputHeaders()

    For lngColumn = 6 To INPUT_COLUMN_COUNT + 5
        If StrComp(strKey, CStr(arrHeaders(lngColumn - 1)), vbTextCompare) = 0 Then
            lngFound = lngColumn
            Exit For
        End If
    Next lngColumn

    If lngFound = 0 And Len(strKey) = 1 Then
        lngColumn = Asc(strKey) - Asc("A") + 1
        If lngColumn >= 6 And lngColumn <= INPUT_COLUMN_COUNT + 5 Then
            lngFound = lngColumn
        End If
    End If

ExitPoint:
    If errNumber = 0 Then ResolveExceptionTransactionColumn = lngFound
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    End If
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-26
' Parameters:    varValue As Variant
' Returns:       String
' Description:   Converts and trims a lookup key using CStr.
'-------------------------------------------------------------------------------
Private Function NormalizeLookupKey(ByVal varValue As Variant) As String
    Const METHOD_NAME As String = "NormalizeLookupKey"
    Dim errDescription As String
    Dim errNumber As Long
    Dim strResult As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    If IsError(varValue) Then
        Call VBA.Err.Raise( _
            ERROR_STATIC_DATA, METHOD_NAME, _
            "An Excel error value cannot be used as a lookup key.")
    End If

    If IsNull(varValue) Or IsEmpty(varValue) Then
        strResult = vbNullString
    Else
        strResult = CStr(Trim$(CStr(varValue)))
    End If

ExitPoint:
    If errNumber = 0 Then NormalizeLookupKey = strResult
    If errNumber <> 0 Then Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Function

Private Function GetOptionalMasterValue(ByVal varValue As Variant) As String
    Const METHOD_NAME As String = "GetOptionalMasterValue"
    Dim errDescription As String

        strResult = Trim$(CStr(varValue))
    End If

ExitPoint:
    If errNumber = 0 Then GetOptionalMasterValue = strResult
    If errNumber <> 0 Then
        Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)

ErrHandler:

    errDescription = VBA.Err.Description

End Function

' Returns:       String
' Description:   Converts a Static_Data value to trimmed text.
'-------------------------------------------------------------------------------
Private Function GetTextValue(ByVal varValue As Variant) As String
    Const METHOD_NAME As String = "GetTextValue"
    Dim errDescription As String
    Dim errNumber As Long
    Dim strResult As String

    If Not DEV_MODE Then On Error GoTo ErrHandler

    If IsError(varValue) Then
        Call VBA.Err.Raise( _
            ERROR_STATIC_DATA, METHOD_NAME, _
            "Static_Data contains an Excel error value.")
    End If

    If IsNull(varValue) Or IsEmpty(varValue) Then
        strResult = vbNullString
    Else
        strResult = Trim$(CStr(varValue))
    End If

ExitPoint:
    If errNumber = 0 Then GetTextValue = strResult
    If errNumber <> 0 Then Call VBA.Err.Raise(errNumber, CLASS_NAME & "." & METHOD_NAME, errDescription)
    Exit Function

ErrHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Call ErrorManager.addError(CLASS_NAME, METHOD_NAME, errNumber, errDescription)
    GoTo ExitPoint
End Function
